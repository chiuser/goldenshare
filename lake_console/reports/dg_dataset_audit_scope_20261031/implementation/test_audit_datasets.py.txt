        assert kwargs == {"temp_policy": "existing_no_spill"}
        assert args[0].temp_directory == Path("/private/tmp")

    yield state
    assert state.violations == [], "Forbidden native reads must not be swallowed."

def target(root, key, coordinate=None):
    template = ENTRIES[key].path_template.removeprefix("data_lake/")
    return root / template.format(
        **({"partition_key": coordinate} if coordinate is not None else {})
    )


def write_parquet(path, schema, rows):
    """Real typed Parquet. Producer schema was checked independently in intake."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with duckdb.connect() as connection:
        columns = ",".join(f'"{c.name}" {c.type}' for c in schema)
        connection.execute(f"CREATE TABLE fixture ({columns})")
        for row in rows:
            names = list(row)
            column_sql = ",".join(f'"{n}"' for n in names)
            values = ",".join("?" for _ in names)
            connection.execute(
                f"INSERT INTO fixture ({column_sql}) VALUES ({values})",
                [row[n] for n in names],
            )
        connection.execute("COPY fixture TO ? (FORMAT PARQUET)", [str(path)])


def write_calendar(
    root,
    *,
    opened=("2026-10-29",),
    start="2026-10-01",
    end="2026-10-31",
    omit=(),
    duplicate=(),
    null=(),
):
    first, last = date.fromisoformat(start), date.fromisoformat(end)
    rows = []
    for index in range((last - first).days + 1):
        day = first + timedelta(days=index)
        key = day.isoformat()
        if key in omit:
            continue
        row = {
            "exchange": "SSE",
            "trade_date": day,
            "is_open": None if key in null else key in opened,
        }
        rows.append(row)
        if key in duplicate:
            rows.append(dict(row))
    path = target(root, "silver_trade_calendar")
    write_parquet(path, ENTRIES["silver_trade_calendar"].column_schema, rows)
    return path


def write_period(
    root, key, dates=("20261029",), *, coordinate=None, bar=None, freq=None
):
    weekly = key in (PERIOD_KEYS[0], PERIOD_KEYS[2])
    coordinate = coordinate or ("2026-10-30" if weekly else "2026-10")
    bar = bar or ("20261030" if weekly else "20261031")
    frequency = freq or ("week" if weekly else "month")
    rows = [
        {
            "ts_code": f"{i + 1:06d}.SZ",
            "trade_date": bar,
            "end_date": d,
            "freq": frequency,
        }
        for i, d in enumerate(dates)
    ]
    path = target(root, key, coordinate)
    write_parquet(path, ENTRIES[key].column_schema, rows)
    return path


def write_announcement(root, day="2026-10-29", rows=None):
    key = "raw_tushare_anns_d"
    path = target(root, key, day)
    if rows is None:
        rows = [
            {
                "ann_date": day.replace("-", ""),
                "ts_code": "000001.SZ",
                "name": None,
                "title": "公告",
                "url": None,
                "rec_time": None,
            }
        ]
    write_parquet(path, ENTRIES[key].column_schema, rows)
    return path


def reports(root, keys, days=("2026-10-29",), observed=OBSERVED):
    entries = tuple(ENTRIES[key] for key in keys)
    return list(
        audit._audit_dates(
            catalog=entries, lake_root=root, trade_dates=days, observed_at=observed
        )
    )


def period_report(root, key, **kwargs):
    return reports(root, [key, "silver_trade_calendar"], **kwargs)[0][
        "period_partition_assets"
    ][key]

@pytest.mark.parametrize(
    "kind",
    [
        "unknown",
        "new_full_model",
        "bad_template",
        "absolute_template",
        "traversal",
        "duplicate",
        "wrong_layout",
    ],
)
def test_unknown_or_conflicting_catalog_does_not_disappear(lake, kind):
    base = ENTRIES["raw_tushare_anns_d"]
    fields = {name: getattr(base, name) for name in base.__dataclass_fields__}
    fields["asset_key"] = "unknown_asset"
    models = list(PARTITION_MODEL_DEFINITIONS)
    if kind == "unknown":
        fields["partition_model"] = "not_registered"
    elif kind == "new_full_model":
        source = next(m for m in models if m.model.value == "full_file_raw_stock_basic")
        models.append(replace(source, model="new_full_file"))
        fields["partition_model"] = "new_full_file"
    elif kind == "bad_template":
        fields["path_template"] = "data_lake/raw/{other}/file.parquet"
    elif kind == "absolute_template":
        fields["path_template"] = (
            "data_lake//outside/ann_date={partition_key}/file.parquet"
        )
    elif kind == "traversal":
        fields["path_template"] = (
            "data_lake/../outside/ann_date={partition_key}/file.parquet"
        )
    elif kind == "wrong_layout":
        source = next(m for m in models if m.model == base.partition_model)
        models[models.index(source)] = replace(source, physical_layout="single_file")
    entry = SimpleNamespace(**fields)
    catalog = (entry, entry) if kind == "duplicate" else (entry,)
    plan = audit._build_audit_plan(
        catalog=catalog, lake_root=lake, trade_dates=("2026-10-29",), models=models
    )
    assert "unknown_asset" in plan["unclassified"]
    assert "unknown_asset" not in plan["excluded"]
    assert plan["catalog_count"] == 1 and plan["issues"]

@pytest.mark.parametrize(
    "kind",
    [
        "bad_date",
        "null_trade_date",
        "wrong_period",
        "bad_freq",
        "null_freq",
        "missing_column",
        "wrong_type",
        "broken",
    ],
)
def test_period_bad_values_versus_unreadable_schema(lake, recording, kind):
    key = PERIOD_KEYS[0]
    write_calendar(lake)
    if kind in {"missing_column", "wrong_type"}:
        schema = [c for c in ENTRIES[key].column_schema if c.name != "end_date"]
        if kind == "wrong_type":
            schema = list(ENTRIES[key].column_schema)
            schema = [
                replace(c, type="DATE") if c.name == "end_date" else c for c in schema
            ]
        path = target(lake, key, "2026-10-30")
        write_parquet(path, schema, [])
    elif kind == "broken":
        path = target(lake, key, "2026-10-30")
        path.parent.mkdir(parents=True)
        path.write_bytes(b"not parquet")
    else:
        dates = ("20260230",) if kind == "bad_date" else ("20261029",)
        bar = "20261023" if kind == "wrong_period" else "20261030"
        path = write_period(
            lake, key, dates, bar=bar, freq="month" if kind == "bad_freq" else None
        )
        if kind in {"null_trade_date", "null_freq"}:
            # Explicit invalid producer input for the regression's negative branch.
            field = "trade_date" if kind == "null_trade_date" else "freq"
            row = {
                "ts_code": "000001.SZ",
                "trade_date": "20261030",
                "end_date": "20261029",
                "freq": "week",
                field: None,
            }
            path.unlink()
            write_parquet(path, ENTRIES[key].column_schema, [row])
    result = period_report(lake, key)
    unreadable = kind in {"missing_column", "wrong_type", "broken"}
    assert result["assessment_status"] == ("unverified" if unreadable else "gap")
    assert result["status"] == ("unreadable_file" if unreadable else "invalid_data")
    assert result["update_status"] == "unverified"
    if unreadable:
        assert result["row_count"] is None

def test_changed_period_invalidates_all_cached_dates_without_retry(lake, recording):
    write_calendar(lake, opened=("2026-10-26", "2026-10-27"))
    path = write_period(lake, PERIOD_KEYS[0], ("20261027",))

    def change(sql, params):
        if "WITH source_rows" in sql:
            current = path.stat()
            os.utime(path, ns=(current.st_atime_ns, current.st_mtime_ns + 1000000000))

    recording.after = change
    result = reports(
        lake,
        [PERIOD_KEYS[0], "silver_trade_calendar"],
        days=("2026-10-26", "2026-10-27"),
    )
    assert all(
        r["period_partition_assets"][PERIOD_KEYS[0]]["assessment_status"]
        == "unverified"
        for r in result
    )
    assert all(
        r["period_partition_assets"][PERIOD_KEYS[0]]["row_count"] is None
        for r in result
    )
    assert sum("WITH source_rows" in sql for sql, _ in recording.queries) == 1

def test_postgres_keeps_forced_readonly_resource_and_date_count(
    lake, recording, monkeypatch
):
    entries = [
        e
        for e in CATALOG
        if e.layer.value == "serving"
        and audit._serving_backend(e) == "prod_postgres"
        and audit._is_trade_date_serving(e)
    ]
    calls = []

    class Cursor:
        def __enter__(self):
            return self

        def __exit__(self, *args):
            pass

        def execute(self, sql, params):
            calls.append((str(sql), params))

        def fetchone(self):
            return (4,)

    class Connection:
        def cursor(self):
            return Cursor()

    @contextmanager
    def readonly(self):
        calls.append("readonly_resource")
        yield Connection()

    monkeypatch.setattr(
        resources.ProdPostgresWriteResource, "connect_readonly", readonly
    )
    monkeypatch.setattr(
        resources.ProdPostgresWriteResource,
        "connect",
        lambda *a: pytest.fail("write connection forbidden"),
    )
    result = audit._audit_prod_postgres_serving(entries, "2026-10-29")
    assert calls[0] == "readonly_resource"
    assert len(calls) == len(entries) + 1
    assert all(
        "SELECT count(*) FROM" in sql
        and "WHERE trade_date = %s" in sql
        and params == ("2026-10-29",)
        for sql, params in calls[1:]
    )
    assert all(
        r["row_count"] == 4 and r["backend"] == "prod_postgres" for r in result.values()
    )

def test_report_name_collision_retries_name_only(tmp_path, monkeypatch):
    output = tmp_path / "reports"
    output.mkdir()
    report = {"trade_date": "2026-10-29"}
    first = audit._publish_report(report, output)
    original = os.link
    attempts = []

    def collision(source, destination):
        attempts.append(destination)
        if len(attempts) == 1:
            # Real existing destination: native link must refuse to overwrite.
            original(source, first)
        return original(source, destination)

    monkeypatch.setattr(os, "link", collision)
    second = audit._publish_report(report, output)
    assert len(attempts) == 2
    assert first != second and json.loads(first.read_text()) == report
    assert not list(output.glob("*.tmp"))
