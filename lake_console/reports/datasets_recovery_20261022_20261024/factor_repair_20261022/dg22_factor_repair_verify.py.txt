import os
import re

import yaml

def identity(path):
    s=path.stat();return [s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns]
def sha_format(value):return isinstance(value,str) and re.fullmatch('[a-f0-9]{64}',value) is not None

os.environ['PGOPTIONS']='-c default_transaction_read_only=on -c statement_timeout=30000'
v=yaml.safe_load((Path(os.environ['DAGSTER_HOME'])/'dagster.yaml').read_text())['storage']['postgres']['postgres_url'];url=os.environ[v['env']] if isinstance(v,dict) else v

    assert r.run_config['ops']['stock_mins_qfq_factor_repair_op']['config']['trade_date']==DAY

    checks=[];failures=[];mats=[]
    for rec in records.records:
        ev=rec.event_log_entry.dagster_event
        if ev.event_type==dg.DagsterEventType.ASSET_CHECK_EVALUATION:

            checks.append({'asset':c.asset_key.to_user_string(),'name':c.check_name,'partition':c.partition,'passed':c.passed,'blocking':c.blocking,'storage_id':rec.storage_id,'metadata':{k.removeprefix('goldenshare/'):v.value for k,v in c.metadata.items()}})

        else:failures.append(str(ev.event_specific_data))
    result.update(run_id=r.run_id,root_run_id=r.root_run_id or r.run_id,status=r.status.value,elapsed_seconds=rr.end_time-rr.start_time if rr.start_time and rr.end_time else None,checks=checks,failures=failures,materializations=mats)

    assert len(checks)==7 and {c['asset'] for c in checks}=={f'gold_stk_mins_qfq_{f}m' for f in FREQS}
    code_hash=digest('\n'.join(sorted(codes)))
    payload={'producer_run_id':r.run_id,'repair_required_codes_hash':code_hash}

    expected={'trade_date':DAY,'previous_trade_date':'2026-10-21','repair_start_trade_date':'2014-01-02','repair_end_trade_date':DAY,'selected_partition_count':3111,'can_execute_repair':True,'repair_required':True,'factor_changed_code_count':26,'repair_required_code_count':26,'repair_required_codes':codes,'repair_required_codes_hash':code_hash,'repair_required_codes_truncated':False,'new_current_code_count':1,'missing_previous_factor_code_count':0,'repaired_code_count':26,'failed_code_count':0,'derived_rewrite_required':True,'derived_repaired_code_count':26,'derived_failed_code_count':0,'failed_row_count':0,'planned_batch_count':65,'executed_batch_count':65,'derived_planned_batch_count':26,'derived_executed_batch_count':26,'producer_run_id':r.run_id,'upstream_batch_id':expected_batch}
    for c in checks:

        assert all(c['metadata'].get(k)==v for k,v in expected.items()),{k:(c['metadata'].get(k),v) for k,v in expected.items() if c['metadata'].get(k)!=v}
        assert c['metadata']==checks[0]['metadata']
    progress={};targets={};sources={};factors={};as_of=[]
    for line in (BASE/'launch.log').read_text().splitlines():

    assert len(targets)==91 and len(sources)==52 and len(factors)==13 and len(as_of)==1

    assert max(int(d['cumulative_stat_bytes']) for d in sources.values())==47307734197
    assert max(int(d['cumulative_stat_bytes']) for d in factors.values())==113188579

    years={str(y):[d for d in manifest['trade_dates'] if d.startswith(str(y))] for y in range(2014,2027)}

                assert not (directory/'operation.json').exists();continue

            paths=operation['targets'];assert len(paths)==len(set(paths))==int(logged['output_file_count'])
