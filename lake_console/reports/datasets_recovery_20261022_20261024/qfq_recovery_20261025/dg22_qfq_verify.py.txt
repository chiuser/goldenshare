import hashlib
import json
import os
import sys
from datetime import datetime
from pathlib import Path
from zoneinfo import ZoneInfo

import dagster as dg
import yaml
from dagster_postgres import PostgresEventLogStorage, PostgresRunStorage
from orchestrator.defs.catalog.lake_assets import LAKE_ASSET_CATALOG

BASE = Path('/Users/congming/github/goldenshare/lake_console/reports/datasets_recovery_20261022_20261024/qfq_recovery_20261025')
ROOT = Path('/Volumes/datasource/data_lake')
STAGE = Path('/Volumes/datasource/data_lake_staging')
DAY = '2026-10-22'
FREQS = (1,5,15,30,60,90,120)
EXPECTED = {f'gold_stk_mins_qfq_{f}m' for f in FREQS}
os.environ['PGOPTIONS'] = '-c default_transaction_read_only=on -c statement_timeout=30000'
v = yaml.safe_load((Path(os.environ['DAGSTER_HOME'])/'dagster.yaml').read_text())['storage']['postgres']['postgres_url']
url = os.environ[v['env']] if isinstance(v,dict) else v
rs = PostgresRunStorage(url, should_autocreate_tables=False)
es = PostgresEventLogStorage(url, should_autocreate_tables=False)

def stat(p):
    s = p.stat()
    return [s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns]

def digest(s):
    return hashlib.sha256(s.encode()).hexdigest()

result = {'date':DAY,'as_of':datetime.now(ZoneInfo('Asia/Shanghai')).isoformat(),'extra_annual_content_reads':0,'extra_sha_passes':0}
try:
    rr = rs.get_run_records(dg.RunsFilter(run_ids=[sys.argv[1]]),limit=1)
    assert len(rr)==1
    rr=rr[0]; r=rr.dagster_run
    assert r.job_name=='stock_mins_qfq_daily_update_job' and r.tags.get('dagster/partition')==DAY
    result.update(run_id=r.run_id,root_run_id=r.root_run_id or r.run_id,status=r.status.value,elapsed_seconds=rr.end_time-rr.start_time if rr.end_time and rr.start_time else None)
    records=es.get_records_for_run(r.run_id,of_type={dg.DagsterEventType.ASSET_MATERIALIZATION,dg.DagsterEventType.ASSET_CHECK_EVALUATION,dg.DagsterEventType.STEP_FAILURE,dg.DagsterEventType.PIPELINE_FAILURE},limit=100)
    assert not records.has_more
    mats=[];checks=[];failures=[]
    for rec in records.records:
        ev=rec.event_log_entry.dagster_event
        if ev.event_type==dg.DagsterEventType.ASSET_MATERIALIZATION:
            m=rec.asset_materialization
            assert m.partition==DAY and m.asset_key.to_user_string() in EXPECTED
            mats.append({'asset':m.asset_key.to_user_string(),'storage_id':rec.storage_id,'metadata':{k:getattr(v,'value',str(v)) for k,v in m.metadata.items() if k!='dagster/column_schema'}})
        elif ev.event_type==dg.DagsterEventType.ASSET_CHECK_EVALUATION:
            c=ev.event_specific_data; t=c.target_materialization_data
            checks.append({'asset':c.asset_key.to_user_string(),'name':c.check_name,'passed':c.passed,'blocking':c.blocking,'target_storage_id':t.storage_id if t else None,'metadata':{k:getattr(v,'value',str(v)) for k,v in c.metadata.items() if k!='dagster/column_schema'}})
        else:
            failures.append(str(ev.event_specific_data))
    result.update(materializations=mats,checks=checks,failures=failures)
    (BASE/'run-verified.json').write_text(json.dumps(result,ensure_ascii=False,indent=2,default=str)+'\n')
    assert r.status==dg.DagsterRunStatus.SUCCESS and not failures
    assert len(mats)==7 and {m['asset'] for m in mats}==EXPECTED
    catalog={str(x.asset_key):x for x in LAKE_ASSET_CATALOG}
    assert len(checks)==28 and {(c['asset'],c['name']) for c in checks}=={(a,n) for a in EXPECTED for n in catalog[a].blocking_check_names}
    mids={m['asset']:m['storage_id'] for m in mats}
    assert all(c['passed'] and c['blocking'] and c['target_storage_id']==mids[c['asset']] for c in checks)
    publications=[]
    for f in FREQS:
        scope={'version':1,'family':'qfq','freq':f,'trade_dates':(DAY,)}
        directory=STAGE/'stk_mins_qfq/qfq'/f'root={digest(str(ROOT.resolve()))}'/f'freq={f}'/f'run_id={digest(result["root_run_id"])}'/f'scope={digest(json.dumps(scope,sort_keys=True))}'
        operation=json.loads((directory/'operation.json').read_text())
        assert operation['scope']==json.loads(json.dumps(scope))
        targets=operation['targets']; assert targets and len(targets)<=5592 and len(set(targets))==len(targets)
        assert len(operation['inputs']['replacement'])==64
        summary={'frequency':f,'directory':str(directory),'files':len(targets),'promoted_files':0,'old_annual_stat_bytes':0,'new_annual_stat_bytes':0,'annual_rows':0}
        for target in targets:
            p=Path(target)
            assert p.is_relative_to(ROOT/'gold/quote/stk_mins_qfq'/f'freq={f}') and p.parent.name=='year=2026' and p.name=='part-000.parquet'
            receipt=json.loads((directory/'files'/f'{digest(target)}.json').read_text())
            assert receipt['status']=='promoted' and receipt['target']==target and len(receipt['sha256'])==64
            candidate=Path(receipt['candidate'])
            assert candidate.is_relative_to(directory/'candidates') and not candidate.exists()
            assert stat(p)[:4]==receipt['candidate_identity'][:4]
            summary['promoted_files']+=1
            summary['old_annual_stat_bytes']+=receipt['before'][2] if receipt['before'] else 0
            summary['new_annual_stat_bytes']+=receipt['candidate_identity'][2]
            summary['annual_rows']+=receipt['row_count']
        publications.append(summary)
    pre=json.loads((BASE/'preflight.json').read_text())
    assert all(stat(Path(p))==s for p,s in pre['inputs'].items())
    result.update(publications=publications,input_stats_unchanged=True,materializations_verified=7,blocking_checks_passed=28,remaining_count_from_scoped_recovery=21)
    (BASE/'run-verified.json').write_text(json.dumps(result,ensure_ascii=False,indent=2,default=str)+'\n')
    print(json.dumps({k:v for k,v in result.items() if k not in ('checks','materializations')},ensure_ascii=False,indent=2))
finally:
    rs.dispose();es.dispose()
