def audit(event, args):
    if event in {"socket.connect", "socket.getaddrinfo"}:
        DENIED.append(event)
        raise AssertionError("Network forbidden")
    if event in {"open", "os.listdir", "os.scandir", "os.mkdir"} and args and isinstance(args[0], (str, bytes, os.PathLike)):
        path = os.fsdecode(args[0])
        if path.startswith(("/Volumes/datasource/", "/Users/congming/.goldenshare/dagster_home", str(DATA / "forbidden-instance"))):
            DENIED.append(event)
            raise AssertionError("Formal/shared resource forbidden")

def path(dataset, code=CODE):
    layer = "quote" if dataset == "stk_mins_qfq" else "indicator"
    return LAKE / "gold" / layer / dataset / "freq=5" / f"ts_code={code}" / "year=2026" / "part-000.parquet"


def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def write_table(connection, name, ddl, rows, target):
    target.parent.mkdir(parents=True, exist_ok=True)
    connection.execute(f"CREATE TABLE {name} ({ddl})")
    connection.executemany(f"INSERT INTO {name} VALUES ({','.join('?' for _ in rows[0])})", rows)
    connection.execute(f"COPY {name} TO ? (FORMAT PARQUET)", [str(target)])
