class PriorVariant(ast.NodeTransformer):
    """Adapt the same assertions to already-existing, exact Oct27 evidence."""
    def visit_Name(self, node):
        if node.id == 'DAY':
            node.id = 'PRIOR'
        return node

    def visit_Call(self, node):
        self.generic_visit(node)
        if isinstance(node.func, ast.Name) and node.func.id == 'target_path':
            if len(node.args) == 1:
                node.args.append(ast.Name(id='PRIOR', ctx=ast.Load()))
        return node

def approved_history_summary(pre,result):
    # The approved 1GiB applies to ALL enumerated 2014..2026 seed/candidate
    # history, deduplicated by path. Normal full-market two-year inputs are
    # separately budgeted; do not exempt candidate 2025/2026 files here.
    inputs={}
    for freq in FREQS:
        key=str(freq)
        groups=(pre['seed_history'][key]['inputs'],result['candidate_groups'][key]['history_inputs'])
        for group in groups:
            for path,value in group.items():
                _,year=source_coordinates(path,freq)
                require(2014<=year<=2026,'History outside approved years')
                if path in inputs:require(inputs[path]['identity']==value,'Conflicting same-path identity')
                inputs[path]={'identity':value,'year':year}
    older={p:v['identity'] for p,v in inputs.items() if v['year']<2025}
    total=sum(v['identity'][2] for v in inputs.values())
    require(total<=1024**3,'Combined seed/candidate history exceeds approved1GiB')
    return {'all_history_files':len(inputs),'all_history_bytes':total,
            'older_files':len(older),'older_bytes':sum(v[2] for v in older.values()),
            'path_deduplicated':True}
