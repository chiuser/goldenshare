def check(condition, message):
    if not condition:
        raise RuntimeError(message)


def write(path, value):
    path.write_text(json.dumps(value), encoding='utf-8')


def update(path, function):
    value = json.loads(path.read_text())
    function(value)
    write(path, value)

def load_module(root, requests, behavior):
    spec = importlib.util.spec_from_file_location('fixture_original_trigger', SOURCE)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    module.OUT = root
    real_path = Path

    def isolated_path(value):
        if str(value) == '/Users/congming/.goldenshare/dagster_logs':
            return root / 'logs'
        return real_path(value)

    module.Path = isolated_path

    def fake_post(url, *, json, timeout):
        # This replaces transport before calling main: no socket is opened.
        check(url == 'http://127.0.0.1:3000/graphql', 'Unexpected URL')
        requests.append(copy.deepcopy(json))
        if json['query'].startswith('mutation Launch'):
            if behavior == 'unknown':
                raise TimeoutError('Synthetic unknown launch outcome')
            payload = {'launchRun': {'__typename': 'LaunchRunSuccess',
                                    'run': {'runId': 'fixture-launched-run', 'status': 'STARTING'}}}
        else:
            check(json['query'].startswith('query R'), 'Unexpected operation')
            payload = {'runOrError': {'__typename': 'Run', 'runId': 'fixture-launched-run',
                                     'status': 'SUCCESS'}}
        return type('Response', (), {'json': lambda self: {'data': payload}})()

    module.requests.post = fake_post
    return module


def invoke(module, stage):
    old = sys.argv
    try:
        sys.argv = [str(SOURCE), '--stage', stage]
        with contextlib.redirect_stdout(io.StringIO()):
            module.main()
        return None
    except Exception as error:
        return error
    finally:
        sys.argv = old


results = []
fixture_roots = []


def case(name, stage, mutate=None, *, allow=False, behavior='success'):
    with tempfile.TemporaryDirectory(prefix='trigger-fixture-', dir=SOURCE.parent) as directory:
        root = Path(directory)
        fixture_roots.append(str(root))
        fixtures(root)
        if mutate:
            mutate(root)
        requests = []
        module = load_module(root, requests, behavior)
        error = invoke(module, stage)
        job, batch, mode, partition = JOBS[stage]
        intent = root / batch / (mode + '-launch-intent.json')
        if allow:
            check(error is None, name + ': ' + repr(error))
            check(len(requests) == 2, name + ': expected one launch and one observation')
            p = requests[0]['variables']['p']
            check(p['selector'] == {'repositoryLocationName': 'orchestrator',
                                   'repositoryName': '__repository__', 'jobName': job}, name)
            check(set(p) == {'selector', 'runConfigData', 'executionMetadata'}, name)
            tags = p['executionMetadata']['tags']
            check({'key': 'dagster/partition', 'value': DAY} in tags if partition
                  else not any(x['key'] == 'dagster/partition' for x in tags), name)
            expected = CONFIG if stage == 'macd-repair' else (
                {'ops': {'stock_mins_qfq_factor_repair_op': {'config': {'trade_date': DAY}}}}
                if stage == 'factor' else {})
            check(p['runConfigData'] == expected, name)
            # A second call must stop before any further GraphQL request.
            second = invoke(module, stage)
            check(second is not None and len(requests) == 2, name + ': duplicate relaunch')
        elif behavior == 'unknown':
            check(isinstance(error, TimeoutError) and len(requests) == 1 and intent.exists(), name)
            second = invoke(module, stage)
            check(second is not None and len(requests) == 1, name + ': unknown launch was resubmitted')
        else:
            check(error is not None, name + ': should refuse')
            check(not requests, name + ': touched GraphQL before rejection')
        results.append({'name': name, 'stage': stage, 'passed': True,
                        'graphql_transport_calls': len(requests),
                        'error': type(error).__name__ if error else None})


def accepted_edit(stage, field, value):

def config_edit(field, value):

        update(root / 'batch10/repair-run-config.json',
               lambda d: d['ops']['gold_stk_mins_qfq_macd_kdj_repair_op']['config'].__setitem__(field, value))
        config = json.loads((root / 'batch10/repair-run-config.json').read_text())

        # rather than merely detecting unequal copies of the same bad config.
        update(root / 'batch10/repair-decision.json', lambda d: d.__setitem__('run_config', config))
    return mutate
