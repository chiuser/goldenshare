    def ready(self, assets):
        return DatasetReadinessStatus(True, tuple(AssetReadinessStatus(asset, self.m.DAY, True, True, True, True,
            100 + number, self.m.DAY, (), (), "isolated") for number, asset in enumerate(assets)))

    def put(self, path, value):
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(value))

    def daily_evidence(self):
        # This is deliberately a stat-only marker, not a Parquet-contract fixture.
        source = self.m.ROOT / "stat-only-source.parquet"
        source.write_bytes(b"isolated-stat-marker")
        frozen = {"date": self.m.DAY, "safe_to_launch": True, "factor_gate": self.factor.to_payload(),
                  "scoped_qfq_inputs": {str(source): self.m.identity(source)},
                  "budget": {"Qr": 100, "Ir_before_daily": 100, "B_before_daily": 1000}}
        self.put(self.m.REPORT / "batch10/preflight.json", frozen)
        daily = {"date": self.m.DAY, "verified": True, "stage_acceptance_complete": True, "status": "SUCCESS",
                 "job": "gold_stk_mins_qfq_macd_kdj_daily_update_job", "run_id": "daily-run", "factor_gate": self.factor.to_payload(),
                 "materializations": [{"asset": asset, "storage_id": 100 + number} for number, asset in enumerate(self.assets)]}
        self.put(self.m.REPORT / "batch10/daily-run-verified.json", daily)
        targets = {}
        for freq in self.m.FREQS:
            state = self.m.ROOT / "gold/indicator/stk_mins_qfq_macd_kdj_state" / ("freq=" + str(freq)) / ("trade_date=" + self.m.DAY) / "part-000.parquet"
            indicator = self.m.ROOT / "gold/indicator/stk_mins_qfq_macd_kdj" / ("freq=" + str(freq)) / "ts_code=600000.SH/year=2026/part-000.parquet"
            targets[str(state)] = {"kind": "state", "identity": [1, 1, 10, 1, 1], "before": None}
            targets[str(indicator)] = {"kind": "indicator", "identity": [1, 1, 20, 1, 1], "before": [1, 1, 15, 1, 1]}
        self.put(self.m.REPORT / "batch10/daily-target-identities.json", targets)
        return source, daily

    def test_original_builder_output_and_budget(self):
        self.daily_evidence()
        result = self.m.prepare_repair()
        expected = {"ops": {"gold_stk_mins_qfq_macd_kdj_repair_op": {"config": {
            "qfq_factor_repair_trade_date": "2026-10-28", "start_trade_date": "2014-01-02", "freqs": [1, 5, 15, 30, 60, 90, 120],
            "stock_codes": ["600000.SH"], "reason": "qfq_factor_repair:2026-10-28", "repair_required_codes_hash": "fixture-hash", "upstream_batch_id": "fixture-batch"}}}}
        self.assertEqual(result["run_config"], expected)
        self.assertEqual(json.loads((self.m.REPORT / "batch10/repair-run-config.json").read_text()), expected)
        self.assertEqual(result["daily_run_id"], "daily-run")
        self.assertEqual(result["repair_budget"]["Ir_after_daily"], 135)
        self.assertEqual(result["repair_budget"]["B_after_daily"], 1070)

    def growth_fixture(self):
        reports = []
        target26 = str(self.m.ROOT / "gold/quote/stk_mins_qfq/freq=1/ts_code=600000.SH/year=2026/part-000.parquet")
        target25 = str(self.m.ROOT / "gold/quote/stk_mins_qfq/freq=1/ts_code=600000.SH/year=2025/part-000.parquet")
        directories = []
        for mode, root_id, scopes in (("daily", "daily-root", (("2026", 10, 15, target26),)),
                                      ("factor", "factor-root", (("2025", 20, 22, target25), ("2026", 15, 18, target26)))):
            report = {"mode": mode, "root_run_id": root_id, "publications": []}
            for year, before_bytes, after_bytes, target in scopes:
                directory = self.m.STAGE / "stk_mins_qfq/qfq/root=fixture/freq=1" / ("run_id=" + self.m.text_hash(root_id)) / ("scope=" + year)
                self.put(directory / "operation.json", {"scope": {"family": "qfq", "freq": 1, "trade_dates": [self.m.DAY if year == "2026" else "2025-01-02"]}, "targets": [target]})
                self.put(directory / "files" / (self.m.text_hash(target) + ".json"), {"target": target, "status": "promoted",
                    "before": [1, 1, before_bytes, 4, 5] if mode == "daily" else [1, 2, before_bytes, 4, 5],
                    "candidate_identity": [1, 2, after_bytes, 4, 5] if mode == "daily" else [1, 3, after_bytes, 4, 5]})
                report["publications"].append({"directory": str(directory), "year": year, "frequency": 1,
                    "promoted_files": 1, "files": 1, "candidate_bytes": after_bytes, "old_annual_bytes": before_bytes})
                directories.append(directory)
            reports.append(report)
        return reports, directories[-1], target26

    def test_qfq_net_growth_includes_2025_and_overlap_once(self):
        reports, _, _ = self.growth_fixture()
        self.assertEqual(self.m.qfq_growth(reports, self.factor.to_payload()), 10)

    def test_qfq_overlap_wrong_before_identity_is_rejected(self):
        reports, directory, target = self.growth_fixture()
        path = directory / "files" / (self.m.text_hash(target) + ".json")
        receipt = json.loads(path.read_text())
        receipt["before"][1] = 999
        self.put(path, receipt)
        with self.assertRaisesRegex(ValueError, "continue"):
            self.m.qfq_growth(reports, self.factor.to_payload())
