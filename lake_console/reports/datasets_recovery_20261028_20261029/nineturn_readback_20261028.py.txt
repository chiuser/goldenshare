def valid_identity(value): return isinstance(value,list) and len(value)==5 and all(type(x) is int for x in value)

def safe_code(code):
    return isinstance(code,str) and bool(code) and code==code.strip() and not any(c in code for c in '/\\*?[]')

def source_coordinates(path,freq):
    parts=Path(path).relative_to(ROOT/'gold/quote/stk_mins_qfq'/f'freq={freq}').parts
    require(len(parts)==3 and parts[0].startswith('ts_code=') and parts[1].startswith('year=') and parts[2]=='part-000.parquet','Unexpected QFQ path')
    code=parts[0].removeprefix('ts_code=');year_text=parts[1].removeprefix('year=')
    require(safe_code(code) and len(year_text)==4 and year_text.isdecimal(),'Unsafe QFQ coordinate')
    return code,int(year_text)

def snap(paths):
    paths=tuple(paths);require(len(paths)==len(set(paths)),'Duplicate input path')
    values={str(p):identity(p) for p in paths}
    require(all(values.values()),'Missing approved input')
    return values

def summary(values):
    return {'files':len(values),'bytes':sum(v[2] for v in values.values())}

def historical_paths(freq,codes):
    # Metadata only; never use the full planner at this boundary.
    root=ROOT/'gold/quote/stk_mins_qfq'/f'freq={freq}'
    paths=tuple(sorted(p for code in codes for p in (root/f'ts_code={code}').glob('year=*/part-000.parquet')))
    for path in paths:
        code,year=source_coordinates(path,freq)
        require(code in codes and 2014<=year<=2026,f'History outside approved years/code: {path}')
    return paths

def guard_group(freq,pre,normal=False,seed_history=False,candidate=None):
    key=str(freq);seed=Path(pre['seed_groups'][key]['path'])
    require(identity(seed)==pre['seed_groups'][key]['identity'],'Seed identity changed')
    if normal:
        expected=pre['normal_paths'][key]
        paths=_minute_nineturn_source_paths(lake_root=ROOT,freq=freq,partition_key=DAY)
        require(tuple(map(str,paths))==tuple(expected),'Normal source file set changed')
        require(snap(paths)==expected,'Normal source identity changed')
    if seed_history:
        expected=pre['seed_history'][key]['inputs']
        paths=historical_paths(freq,tuple(pre['factor_status']['repair_required_codes']))
        require(snap(paths)==expected,'Actual changed-code history identity changed')
    if candidate is not None:
        expected=candidate['history_inputs']
        paths=historical_paths(freq,tuple(candidate['candidate_codes']))
        require(snap(paths)==expected,'Actual candidate history identity changed')

def compare_sql(paths: tuple[Path, ...], seed: Path, freq: int, codes: tuple[str, ...]) -> str:
    # Reuse the original full-history formula.  The prior-date filter is outside
    # that query, after its LAG/count calculation; never clip its price inputs.
    full_history = build_gold_stk_mins_qfq_nineturn_select_sql(
        source_paths=paths, freq=freq, stock_codes=codes
    )
    code_values = ", ".join(f"({duckdb_string(code)})" for code in codes)
    code_filter = ", ".join(duckdb_string(code) for code in codes)
    columns = ", ".join(COMPARE_FIELDS)
    output_columns = ",\n      ".join(
        f"{side}.{column} AS {side}_{column}"
        for side in ("expected", "seed")
        for column in COMPARE_FIELDS
    )
    return f"""
    WITH expected AS (
      SELECT ts_code, {columns}
      FROM ({full_history})
      WHERE trade_date = DATE {duckdb_string(PRIOR)}
      QUALIFY row_number() OVER (PARTITION BY ts_code ORDER BY trade_time DESC) = 1
    ), seed AS (
      SELECT CAST(ts_code AS VARCHAR) AS ts_code,
        CAST(trade_date AS DATE) AS trade_date,
        CAST(trade_time AS TIMESTAMP) AS trade_time,
        CAST(up_count AS INTEGER) AS up_count,
        CAST(down_count AS INTEGER) AS down_count,
        CAST(nine_up_turn AS VARCHAR) AS nine_up_turn,
        CAST(nine_down_turn AS VARCHAR) AS nine_down_turn
      FROM {read_parquet(seed, hive_partitioning=False)}
      WHERE CAST(freq AS INTEGER) = {freq}
        AND CAST(trade_date AS DATE) = DATE {duckdb_string(PRIOR)}
        AND CAST(ts_code AS VARCHAR) IN ({code_filter})
      QUALIFY row_number() OVER (
        PARTITION BY ts_code ORDER BY CAST(trade_time AS TIMESTAMP) DESC
      ) = 1
    ), codes AS (
      SELECT ts_code::VARCHAR AS ts_code
      FROM (VALUES {code_values}) AS approved_codes(ts_code)
    )
    SELECT codes.ts_code,
      {output_columns}
    FROM codes
    LEFT JOIN expected USING (ts_code)
    LEFT JOIN seed USING (ts_code)
    ORDER BY codes.ts_code
    """

def verify_events(run_id, preflight, output):
    require(Path(os.environ.get('DAGSTER_HOME', '')) == APPROVED_DAGSTER_HOME, 'DAGSTER_HOME must match the approved formal instance.')
    os.environ['PGOPTIONS'] = '-c default_transaction_read_only=on -c statement_timeout=30000'
    value = yaml.safe_load((APPROVED_DAGSTER_HOME / 'dagster.yaml').read_text())['storage']['postgres']['postgres_url']
    url = os.environ[value['env']] if isinstance(value, dict) else value
    runs = PostgresRunStorage(url, should_autocreate_tables=False)
    events = PostgresEventLogStorage(url, should_autocreate_tables=False)
    try:
        found = runs.get_run_records(dg.RunsFilter(run_ids=[run_id]), limit=1)
        require(len(found) == 1, 'Exact run ID not found.')
        record = found[0]
        run = record.dagster_run
        require(run.job_name == JOB and run.tags.get('dagster/partition') == DAY, 'Wrong original job or partition.')
        require(record.create_timestamp.timestamp() >= datetime.fromisoformat(preflight['as_of']).timestamp(), 'Run predates this preflight.')
        output.update(run_id=run.run_id, root_run_id=run.root_run_id or run.run_id,
                      job=run.job_name, status=run.status.value,
                      elapsed_seconds=record.end_time-record.start_time if record.end_time and record.start_time else None)
        require(run.status == dg.DagsterRunStatus.SUCCESS, 'Run has not completed successfully.')
        result = events.get_records_for_run(
            run_id, of_type={dg.DagsterEventType.ASSET_MATERIALIZATION,
                             dg.DagsterEventType.ASSET_CHECK_EVALUATION,
                             dg.DagsterEventType.STEP_FAILURE,
                             dg.DagsterEventType.PIPELINE_FAILURE}, limit=100,
        )
        require(not result.has_more, 'Unexpected event pagination; do not truncate evidence.')
        mats, checks = [], []
        for entry in result.records:
            event = entry.event_log_entry.dagster_event
            if event.event_type == dg.DagsterEventType.ASSET_MATERIALIZATION:
                item = entry.asset_materialization
                mats.append({'asset': item.asset_key.to_user_string(), 'storage_id': entry.storage_id,
                             'partition': item.partition, 'metadata': metadata(item.metadata)})
            elif event.event_type == dg.DagsterEventType.ASSET_CHECK_EVALUATION:
                item = event.event_specific_data
                target = item.target_materialization_data
                checks.append({'asset': item.asset_key.to_user_string(), 'name': item.check_name,
                               'storage_id': entry.storage_id, 'passed': item.passed,
                               'blocking': item.blocking, 'partition': item.partition,
                               'target_storage_id': target.storage_id if target else None,
                               'target_run_id': target.run_id if target else None,
                               'metadata': metadata(item.metadata)})
            else:
                raise ValueError('Run contains a failure event.')
        expected = {f'gold_stk_mins_qfq_nineturn_{freq}m' for freq in FREQS}
        require(len(mats) == 4 and {x['asset'] for x in mats} == expected, 'Expected exactly four original materializations.')
        require(len(checks) == 4 and {x['asset'] for x in checks} == expected, 'Expected exactly four original checks.')
        by_asset = {x['asset']: x for x in mats}
        for freq in FREQS:
            name = f'gold_stk_mins_qfq_nineturn_{freq}m'
            materialization = by_asset[name]
            require(materialization['partition'] == DAY, 'Materialization has wrong date.')
            item = materialization['metadata']
            count = ROWS[freq]
            require(item['dagster/row_count'] == item['goldenshare/source_row_count'] == count, 'Materialization row counts differ from approved source evidence.')
            require(item['dagster/uri'] == str(target_path(freq)), 'Materialization has wrong formal path.')
            require(item['goldenshare/observed_columns'] == COLUMNS, 'Unexpected observed column order.')
            require(item['goldenshare/formula_version'] == QFQ_NINETURN_VERSION and item['goldenshare/result_status'] == 'written', 'Unexpected formula version or result status.')
            require(valid_digest(item['goldenshare/source_fingerprint']), 'Invalid business source fingerprint metadata.')
            input_summary = item['goldenshare/input_summary']
            filter_summary = item['goldenshare/filter_summary']
            require(input_summary['partition_key'] == DAY and input_summary['freq'] == freq,
                    'Input summary has wrong date or frequency.')
            require(input_summary['source_row_count'] == filter_summary['output_row_count'] == count, 'Nested row counts differ.')
            require(type(input_summary['source_file_count']) is int and input_summary['source_file_count'] > 0, 'Invalid source file count.')
            stock_count = item['goldenshare/stock_code_count']
            fallback_count = item['goldenshare/fallback_recomputed_code_count']
            require(type(stock_count) is int and 0 < stock_count <= count, 'Invalid stock count.')
            require(type(fallback_count) is int and 0 <= fallback_count <= min(QFQ_NINETURN_FALLBACK_CODE_LIMIT, stock_count), 'Invalid fallback count.')
            require(filter_summary['stock_code_count'] == stock_count and filter_summary['fallback_recomputed_code_count'] == fallback_count, 'Nested stock/fallback counts differ.')
            check = next(x for x in checks if x['asset'] == name)
            require(check['passed'] and check['blocking'] and check['partition'] == DAY,
                    'Original blocking check did not pass for target date.')
            require(check['name'] == name + '_integrity_check', 'Unexpected check name.')
            require(check['target_storage_id'] == materialization['storage_id'] and check['target_run_id'] == run_id,
                    'Check is not bound to this run materialization.')
            detail = check['metadata']
            require(detail['goldenshare/file_path'] == str(target_path(freq)), 'Check has wrong formal path.')
            require(detail['goldenshare/checked_row_count'] == count and detail['goldenshare/failed_row_count'] == 0, 'Check row counts are not clean.')
            require(detail['goldenshare/failed_rule_names'] == [], 'Check contains failed rules.')
            summary = detail['goldenshare/rule_summary']
            require(summary['rules'] == RULES and summary['checked_row_count'] == summary['source_row_count'] == count, 'Check rule/source counts differ.')
            require(all(summary[key] == 0 for key in ('source_duplicate_key_count', 'duplicate_key_count', 'null_key_count', 'invalid_value_count', 'missing_source_key_count', 'extra_output_key_count')), 'Check diagnostics contain failures.')
        output.update(materializations=mats, checks=checks, materializations_verified=4, blocking_checks_passed=4)
        return by_asset
    finally:
        runs.dispose()
        events.dispose()

def verify_publications(root_run_id, by_asset):
    result = []
    for freq in FREQS:
        scope = {'version': 1, 'family': 'nineturn', 'freq': freq, 'trade_dates': (DAY,)}
        directory = STAGE / 'stk_mins_qfq/nineturn' / f'root={text_hash(str(ROOT))}' / f'freq={freq}' / f'run_id={text_hash(root_run_id)}' / f'scope={text_hash(json.dumps(scope, sort_keys=True))}'
        target = target_path(freq)
        operation = json.loads((directory / 'operation.json').read_text())
        require(operation['scope'] == json.loads(json.dumps(scope)), 'Publication scope mismatch.')
        require(operation['targets'] == [str(target)], 'Publication must contain only the one target.')
        require(set(operation['inputs']) == {'nineturn_inputs'} and valid_digest(operation['inputs']['nineturn_inputs']), 'Publication input description is missing.')
        receipt_path = directory / 'files' / f'{text_hash(str(target))}.json'
        require(sorted((directory / 'files').glob('*.json')) == [receipt_path], 'Unexpected publication receipt set.')
        receipt = json.loads(receipt_path.read_text())
        require(receipt['target'] == str(target) and receipt['status'] == 'promoted', 'Target is not recorded as promoted.')
        require(receipt['before'] is None, 'This recovery unexpectedly overwrote an existing target.')
        require(valid_digest(receipt['sha256']), 'Invalid recorded candidate digest.')
        require(valid_identity(receipt['candidate_identity']) and valid_identity(receipt['promoted_identity']), 'Invalid recorded candidate/formal identity.')
        observed = identity(target)
        require(observed == receipt['promoted_identity'] and observed[:4] == receipt['candidate_identity'][:4], 'Formal file no longer matches its promotion record.')
        require(receipt['row_count'] == by_asset[f'gold_stk_mins_qfq_nineturn_{freq}m']['metadata']['dagster/row_count'] == ROWS[freq], 'Receipt and materialization row counts differ.')
        candidate = Path(receipt['candidate'])
        require(candidate.is_relative_to(directory / 'candidates') and '..' not in candidate.parts, 'Candidate path escaped its publication.')
        require(candidate.relative_to(directory / 'candidates').parts[1:] == target.relative_to(ROOT).parts,
                'Candidate path does not match target suffix.')
        require(identity(candidate) is None, 'Promoted candidate still exists.')
        result.append({'freq': freq, 'directory': str(directory), 'target': str(target),
                       'row_count': receipt['row_count'], 'bytes': observed[2],
                       'status': receipt['status'], 'formal_identity': observed,
                       'recorded_sha256': receipt['sha256'], 'new_data_sha_reads': 0})
    return result

            require(check['target_storage_id'] == materialization['storage_id'] and check['target_run_id'] == run_id, 'Check is not bound to this run materialization.')
            detail = check['metadata']
            require(detail['goldenshare/file_path'] == str(target_path(freq, PRIOR)), 'Check has wrong formal path.')
            require(detail['goldenshare/checked_row_count'] == count and detail['goldenshare/failed_row_count'] == 0, 'Check row counts are not clean.')
            require(detail['goldenshare/failed_rule_names'] == [], 'Check contains failed rules.')
            summary = detail['goldenshare/rule_summary']
            require(summary['rules'] == RULES and summary['checked_row_count'] == summary['source_row_count'] == count, 'Check rule/source counts differ.')
            require(all((summary[key] == 0 for key in ('source_duplicate_key_count', 'duplicate_key_count', 'null_key_count', 'invalid_value_count', 'missing_source_key_count', 'extra_output_key_count'))), 'Check diagnostics contain failures.')
        output.update(materializations=mats, checks=checks, materializations_verified=4, blocking_checks_passed=4)
        return by_asset
    finally:
        runs.dispose()
        events.dispose()

def verify_prior_publications(root_run_id, by_asset):
    result = []
    for freq in FREQS:
        scope = {'version': 1, 'family': 'nineturn', 'freq': freq, 'trade_dates': (PRIOR,)}
        directory = STAGE / 'stk_mins_qfq/nineturn' / f'root={text_hash(str(ROOT))}' / f'freq={freq}' / f'run_id={text_hash(root_run_id)}' / f'scope={text_hash(json.dumps(scope, sort_keys=True))}'
        target = target_path(freq, PRIOR)
        operation = json.loads((directory / 'operation.json').read_text())
        require(operation['scope'] == json.loads(json.dumps(scope)), 'Publication scope mismatch.')
        require(operation['targets'] == [str(target)], 'Publication must contain only the one target.')
        require(set(operation['inputs']) == {'nineturn_inputs'} and valid_digest(operation['inputs']['nineturn_inputs']), 'Publication input description is missing.')
        receipt_path = directory / 'files' / f'{text_hash(str(target))}.json'
        require(sorted((directory / 'files').glob('*.json')) == [receipt_path], 'Unexpected publication receipt set.')
        receipt = json.loads(receipt_path.read_text())
        require(receipt['target'] == str(target) and receipt['status'] == 'promoted', 'Target is not recorded as promoted.')
        require(receipt['before'] is None or valid_identity(receipt['before']), 'Invalid prior before identity.')
        require(valid_digest(receipt['sha256']), 'Invalid recorded candidate digest.')
        require(valid_identity(receipt['candidate_identity']) and valid_identity(receipt['promoted_identity']), 'Invalid recorded candidate/formal identity.')
        observed = identity(target)
        require(observed == receipt['promoted_identity'] and observed[:4] == receipt['candidate_identity'][:4], 'Formal file no longer matches its promotion record.')
        require(receipt['row_count'] == by_asset[f'gold_stk_mins_qfq_nineturn_{freq}m']['metadata']['dagster/row_count'] == by_asset[f'gold_stk_mins_qfq_nineturn_{freq}m']['metadata']['goldenshare/source_row_count'], 'Receipt and materialization row counts differ.')
        candidate = Path(receipt['candidate'])
        require(candidate.is_relative_to(directory / 'candidates') and '..' not in candidate.parts, 'Candidate path escaped its publication.')
        require(candidate.relative_to(directory / 'candidates').parts[1:] == target.relative_to(ROOT).parts, 'Candidate path does not match target suffix.')
        require(identity(candidate) is None, 'Promoted candidate still exists.')
        result.append({'freq': freq, 'directory': str(directory), 'target': str(target), 'row_count': receipt['row_count'], 'bytes': observed[2], 'status': receipt['status'], 'formal_identity': observed, 'recorded_sha256': receipt['sha256'], 'new_data_sha_reads': 0})
    return result

def controls(expected=None):

    url=postgres_url()

        factor_report=current_accepted(BASE.parent/'batch9/factor-run-verified.json','stock_mins_qfq_factor_repair_job','factor')

            require(len(latest)==1 and latest[0].storage_id==materialization['storage_id'],'MACD daily materialization changed')
        repair_run_id=None
        if gate['requires_macd_kdj_repair']:
            repair=current_accepted(BASE.parent/'batch10/repair-run-verified.json','gold_stk_mins_qfq_macd_kdj_repair_job','repair')
            require(repair['factor_gate']==factor and repair['full_gate']==gate,'MACD accepted repair/full gate mismatch')
            require(repair['completion_gate']==gate['macd_kdj_repair_status'],'MACD completion differs from current full gate')
            require(len(repair['checks'])==2*len(STK_MINS_QFQ_FREQS) and {x['storage_id'] for x in repair['checks']}==set(gate['macd_kdj_repair_status']['event_storage_ids']),'MACD14 completion evidence differs')
            repair_run_id=repair['run_id']
        else:
            require(not factor['rewrote_history'],'Rewritten QFQ history cannot skip MACD repair')
        active_statuses=[dg.DagsterRunStatus.NOT_STARTED,dg.DagsterRunStatus.QUEUED,dg.DagsterRunStatus.STARTING,dg.DagsterRunStatus.STARTED,dg.DagsterRunStatus.CANCELING]
        for job in WRITER_JOBS:
            require(not runs.get_runs(dg.RunsFilter(job_name=job,statuses=active_statuses),limit=1),f'Related writer active: {job}')
        pool=events.get_concurrency_info('gold_stk_mins_qfq_writer')

        return {'factor_status':factor,'macd_complete_gate':gate,'previous_registered_day':PRIOR,
                'factor_run_id':factor_report['run_id'],'macd_daily_run_id':daily['run_id'],'macd_repair_run_id':repair_run_id,

    finally:runs.dispose();events.dispose();schedules.dispose()

def approved_history_summary(pre,result):
    # The approved 1GiB applies to ALL enumerated 2014..2026 seed/candidate
    # history, deduplicated by path. Normal full-market two-year inputs are
    # separately budgeted; do not exempt candidate 2025/2026 files here.
    inputs={}
    for freq in FREQS:
        key=str(freq)
        groups=(pre['seed_history'][key]['inputs'],result['candidate_groups'][key]['history_inputs'])
        for group in groups:
            for path,value in group.items():
                _,year=source_coordinates(path,freq)
                require(2014<=year<=2026,'History outside approved years')
                if path in inputs:require(inputs[path]['identity']==value,'Conflicting same-path identity')
                inputs[path]={'identity':value,'year':year}
    older={p:v['identity'] for p,v in inputs.items() if v['year']<2025}
    total=sum(v['identity'][2] for v in inputs.values())
    require(total<=1024**3,'Combined seed/candidate history exceeds approved1GiB')
    return {'all_history_files':len(inputs),'all_history_bytes':total,
            'older_files':len(older),'older_bytes':sum(v[2] for v in older.values()),
            'path_deduplicated':True}


def run_prepare():
    approved_history()
    pre=load(BASE/'metadata-preflight.json');require(pre['passed'] and pre['date']==DAY,'Incomplete metadata')
    require(not (BASE/'candidate-scope.json').exists(),'Preserve prior queries; do not rerun automatically')
    controls(expected=pre)
    result={'as_of':now(),'date':DAY,'prior_date':PRIOR,'passed':False,'phase':'start','status':'running','formal_writes':0,'jobs_launched':0,'full_planner_calls':0,'fallback_history_queries':0,'new_data_sha_reads':0,'seed_comparisons':{},'candidate_groups':{},'actual_factor_codes':pre['factor_status']['repair_required_codes'],'factor_batch':pre['factor_status']['upstream_batch_id']}
    save('candidate-scope.json',result,new=True)
    try:
        with DuckDBResource().connect() as connection:
            # Key-only normal-source queries precede ALL earlier-history reads.
            for freq in FREQS:
                key=str(freq);result['phase']=f'bounded_candidate_{freq}';save('candidate-scope.json',result)
                guard_group(freq,pre,normal=True)
                count,candidates=_bounded_candidate_codes(connection,source_paths=tuple(map(Path,pre['normal_paths'][key])),partition_key=DAY,previous_partition_path=Path(pre['seed_groups'][key]['path']),freq=freq,after_query=lambda f=freq:guard_group(f,pre,normal=True))
                require(candidates==tuple(sorted(set(candidates))) and all(map(safe_code,candidates)),'Unsafe candidate list')
                paths=historical_paths(freq,candidates);values=snap(paths)
                result['candidate_groups'][key]={'source_row_count':count,'candidate_codes':list(candidates),'candidate_code_count':len(candidates),'history_inputs':values,'history_scope_complete':True,'history_files':len(values),'history_bytes':sum(v[2] for v in values.values())}
                save('candidate-scope.json',result)
            require(set(result['candidate_groups'])==set(map(str,FREQS)),'Candidate frequencies incomplete')
            result['history_scope']=approved_history_summary(pre,result)
            result['candidate_history_files']=sum(g['history_files'] for g in result['candidate_groups'].values())
            result['candidate_history_bytes']=sum(g['history_bytes'] for g in result['candidate_groups'].values())
            result['phase']='all_history_scope_verified';save('candidate-scope.json',result)
            codes=tuple(pre['factor_status']['repair_required_codes'])
            if pre['factor_status']['rewrote_history']:
                require(codes,'Rewrote-history scope has no actual codes')
                for freq in FREQS:
                    key=str(freq);result['phase']=f'seed_compare_{freq}';save('candidate-scope.json',result)
                    guard_group(freq,pre,seed_history=True,candidate=result['candidate_groups'][key])
                    paths=tuple(map(Path,pre['seed_history'][key]['inputs']))
                    require(paths,'Rewrote-history seed source set missing')
                    rows=connection.execute(compare_sql(paths,Path(pre['seed_groups'][key]['path']),freq,codes)).fetchall()
                    guard_group(freq,pre,seed_history=True,candidate=result['candidate_groups'][key])
                    require(len(rows)==len(codes),'Unexpected seed comparison count')
                    differences=[];both_absent=0
                    for row in rows:
                        expected=dict(zip(COMPARE_FIELDS,row[1:7],strict=True));seed=dict(zip(COMPARE_FIELDS,row[7:13],strict=True))
                        both_absent+=expected['trade_time'] is None and seed['trade_time'] is None
                        if expected!=seed:differences.append({'ts_code':row[0],'expected':expected,'seed':seed})
                    result['seed_comparisons'][key]={'result_count':len(rows),'matches':len(rows)-len(differences),'both_absent_on_prior_date':both_absent,'mismatch_count':len(differences),'differences':differences}
                    save('candidate-scope.json',result)
                    require(not differences,'Changed-code27 seed differs; stop before original job')
            else:
                result['seed_check_skipped_reason']='Native factor status rewrote_history=false; exact27 seed receipt/stat reused'
        result.update(passed=True,status='passed_seed_and_candidate_scope',phase='complete',updated_at=now())
    except Exception as error:
        result.update(status='failed_without_retry',failure_type=type(error).__name__,failure=str(error) if isinstance(error,ValueError) else 'Query/stat failed; inspect error without exporting price data')
        raise
    finally:save('candidate-scope.json',result)
    print(json.dumps({'mode':'prepare','passed':result['passed'],'history_scope':result['history_scope'],'candidate_counts':{k:v['candidate_code_count'] for k,v in result['candidate_groups'].items()},'source_rows':{k:v['source_row_count'] for k,v in result['candidate_groups'].items()}}))

def run_verify(run_id):
    global ROWS
    pre=load(BASE/'metadata-preflight.json');candidate=load(BASE/'candidate-scope.json')
    intent=load(BASE/'daily-launch-intent.json');native=load(BASE/'daily-native-run.json')
    require(pre['passed'] and candidate['passed'] and pre['date']==candidate['date']==DAY and pre['previous_registered_day']==PRIOR,'Incomplete preflight/candidate scope')
    require(candidate['prior_date']==PRIOR and candidate['factor_batch']==pre['factor_status']['upstream_batch_id'] and candidate['actual_factor_codes']==pre['factor_status']['repair_required_codes'],'Candidate scope belongs to a different predecessor batch')
    require(native['run_id']==run_id and native['status']=='SUCCESS' and native['trade_date']==DAY,'Wrong current native run')
    require(intent['run_id']==run_id and intent['stage']=='nineturn' and intent['trade_date']==DAY,'Wrong current launch intent')
    require(intent['request']==native['request'] and intent['request']['selector']['jobName']==JOB and intent['request']['runConfigData']=={},'Wrong original launch request')
    require({'key':'dagster/partition','value':DAY} in intent['request']['executionMetadata']['tags'],'Wrong launch partition')
    ROWS={freq:candidate['candidate_groups'][str(freq)]['source_row_count'] for freq in FREQS}
    require(all(type(n) is int and n>0 for n in ROWS.values()),'Missing real source counts')
    require(all(pre['neighbor_stat'][str(target_path(freq))] is None for freq in FREQS),'Targets were not empty before launch')
    output={'as_of':now(),'date':DAY,'passed':False,'formal_content_reads':0,'new_data_sha_reads':0,'formal_writes':0,'verification':'original exact run/checks plus publication JSON/stat; counts from current bounded candidate query','read_scope_evidence':'candidate-scope.json','native_job_reexecutions':0}
    require(not (BASE/'final-verified.json').exists(),'Preserve original final receipt')
    try:
        by_asset=verify_events(run_id,{'as_of':intent['submitted_at']},output)
        output['publications']=verify_publications(output['root_run_id'],by_asset)
        output['input_identity_acceptance']=complete_inputs(pre,candidate)
        output['neighbor_stat']=verify_neighbors(pre)
        output['controls_final']=controls(expected=pre)
        output['output_rows']=sum(x['row_count'] for x in output['publications'])
        output['output_bytes']=sum(x['bytes'] for x in output['publications'])
        require(output['output_rows']==sum(ROWS.values()),'Total rows differ from actual current query')
        output.update(passed=True,stage_acceptance_complete=True)
    except Exception as error:
        output.update(failure_type=type(error).__name__,failure=str(error) if isinstance(error,ValueError) else 'Verification failed; inspect process error')
        raise
    finally:save('final-verified.json',output,new=True)
    print(json.dumps({key:output[key] for key in ('run_id','status','passed','materializations_verified','blocking_checks_passed','output_rows','output_bytes')}))


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('mode',choices=('metadata','prepare','before','verify'))
    parser.add_argument('--run-id')
    args=parser.parse_args()
    if args.mode=='metadata':run_metadata()
    elif args.mode=='prepare':run_prepare()
    elif args.mode=='before':run_before()
    else:
        if not args.run_id:parser.error('verify requires exact --run-id')
        run_verify(args.run_id)


if __name__=='__main__':main()
