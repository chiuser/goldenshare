"""Exact native checks and receipt/stat acceptance, no additional price audit."""
import argparse
import hashlib
import importlib.machinery
import importlib.util
import json
import shutil
from pathlib import Path

from common import BASE, DAY, OUT, ROOT, exact_accept, load, no_conflict, now, require, save, stores

ASSETS = {
    "raw60": ({"raw_stk_mins_60m"}, 3),
    "silver60": ({"silver_stk_mins_60m"}, 4),
    "qfq60_120": ({"gold_stk_mins_qfq_60m", "gold_stk_mins_qfq_120m"}, 8),
    "turnover": ({"gold_wealth_market_turnover", "prod_core_wealth_market_turnover"}, 1),
}


def helper():
    path = BASE.parent / "datasets_recovery_20261028_20261029/qfq_factor_readback_20261028.py.txt"
    loader = importlib.machinery.SourceFileLoader("qfq_existing_readback", str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    module.DAY = DAY
    module.REPORT_ROOT = BASE
    return module


def identity(path):
    value = path.stat()
    return [value.st_dev, value.st_ino, value.st_size, value.st_mtime_ns, value.st_ctime_ns]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("stage", choices=ASSETS)
    stage = parser.parse_args().stage
    directory = OUT / stage
    assets, count = ASSETS[stage]
    result = exact_accept(directory, assets, count)
    if stage in {"raw60", "silver60"}:
        mat = result["materializations"][0]
        require(mat["metadata"]["goldenshare/write_mode"] == "replace_existing", "Wrong mode")
        target = Path(mat["metadata"]["dagster/uri"])
        dataset = "raw_stk_mins" if stage == "raw60" else "silver_stk_mins"
        receipt = Path('/Volumes/datasource/data_lake_staging/minute_partitions') / dataset / (
            "target=" + hashlib.sha256(str(target.resolve()).encode()).hexdigest()) / (
            "run_id=" + hashlib.sha256(result["root_run_id"].encode()).hexdigest()) / "receipt.json"
        value = load(receipt)
        require(value["status"] == "promoted" and value["target"] == str(target)
                and not Path(value["candidate"]).exists(), "Minute publication not complete")
        require(value["result"]["row_count"] == mat["metadata"]["dagster/row_count"], "Receipt/result count mismatch")
        if stage == "raw60":
            require(mat["metadata"]["dagster/row_count"] == 27885, "Raw does not match approved source count")
        result["publication"] = {"receipt": str(receipt), "status": value["status"], "target_stat": identity(target)}
        save(directory / "daily-run-verified.json", result)
    if stage == "raw60":
        save(OUT / "silver60/run-config.json", {"ops": {"silver_stk_mins_60m": {"config": {"write_mode": {"replace_existing": {}}}}}})
        save(OUT / "silver60/preflight.json", {"date": DAY, "safe_to_launch": True, "upstream_run_id": result["run_id"], "observed_at": now()})
    if stage == "silver60":
        require(shutil.disk_usage('/Volumes/datasource/data_lake_staging').free > 128 * 1024**3, "Approved working space unavailable")
        from orchestrator.defs.paths import silver_adj_factor_path, silver_stk_mins_path, silver_stock_basic_path, silver_stock_lifecycle_path, silver_trade_calendar_path
        paths = [silver_stk_mins_path(ROOT, f, DAY) for f in (1, 5, 15, 30, 60)] + [silver_adj_factor_path(ROOT, d) for d in ("2026-10-29", DAY)] + [silver_stock_basic_path(ROOT), silver_stock_lifecycle_path(ROOT), silver_trade_calendar_path(ROOT)]
        inputs = {str(p): identity(p) for p in paths}
        save(OUT / "qfq60_120/run-config.json", {})
        save(OUT / "qfq60_120/preflight.json", {"date": DAY, "safe_to_launch": True,
             "upstream_silver60_run_id": result["run_id"], "other_silver_run_id": "ae0067a5-e040-449d-a6ac-0f32b943788f", "inputs": inputs, "observed_at": now()})
        save(OUT / "turnover/run-config.json", {})
        save(OUT / "turnover/preflight.json", {"date": DAY, "safe_to_launch": True, "upstream_silver60_run_id": result["run_id"], "observed_at": now()})
    if stage == "qfq60_120":
        module = helper()
        logs = list((Path('/Users/congming/.goldenshare/dagster_logs') / result["run_id"] / "compute_logs").glob("*.out"))
        require(logs, "Missing original compute logs")
        progress, targets, sources, factors, as_of = module.load_logs(logs)
        require(not targets and not sources and not factors and not as_of, "Wrong computation logs")
        publications = [module.verify_operation(d, value, result["root_run_id"], "daily", None) for d, value in progress.items()]
        require(len(publications) == 2 and {x["frequency"] for x in publications} == {60, 120}, "Expected exactly two QFQ completed scopes")
        before = load(directory / "preflight.json")["inputs"]
        require(before == {p: identity(Path(p)) for p in before}, "QFQ inputs changed during selected recovery")
        result["publications"] = publications
        save(directory / "daily-run-verified.json", result)
        kept = {f"gold_stk_mins_qfq_{f}m" for f in (1, 5, 15, 30, 90)}
        with stores() as (rs, es):
            import dagster as dg
            old_id = "df08f87b-2316-4368-a9f1-6771bd9c17fd"
            records = rs.get_run_records(dg.RunsFilter(run_ids=[old_id]), limit=1)
            require(len(records) == 1 and records[0].dagster_run.status == dg.DagsterRunStatus.FAILURE, "Original partial run identity changed")
            old_mats, old_checks, _ = module.events_for_run(es, old_id)
            mats = [x for x in old_mats if x["asset"] in kept]
            checks = [x for x in old_checks if x["asset"] in kept]
            require(len(mats) == 5 and len(checks) == 20, "Unaffected QFQ coverage incomplete")
            ids = {x["asset"]: x["storage_id"] for x in mats}
            require(all(x["partition"] == DAY for x in mats) and all(
                x["passed"] and x["blocking"] and x["partition"] in (None, DAY)
                and x["target_storage_id"] == ids[x["asset"]] for x in checks), "Original unaffected checks not bound/passing")
        composite = {"date": DAY, "verified": True, "status": "COMPLETE_BY_ASSET", "stage_acceptance_complete": True,
                     "old_run_id": old_id, "old_run_status": "FAILURE", "new_run_id": result["run_id"],
                     "kept_frequencies": [1, 5, 15, 30, 90], "new_frequencies": [60, 120],
                     "old_materializations": mats, "old_checks": checks,
                     "new_materializations": result["materializations"], "new_checks": result["checks"],
                     "asset_count": 7, "current_check_count": 28}
        save(OUT / "qfq-current-seven-accepted.json", composite)
        save(OUT / "factor/run-config.json", {"ops": {"stock_mins_qfq_factor_repair_op": {"config": {"trade_date": DAY}}}})
        save(OUT / "factor/preflight.json", {"date": DAY, "safe_to_launch": True, "qfq_accepted": "qfq-current-seven-accepted.json", "observed_at": now()})
    save(directory / "daily-run-verified.json", result)
    save(directory / "control-after-acceptance.json", no_conflict())
    print(json.dumps({"stage": stage, "run_id": result["run_id"], "verified": True,
                      "assets": len(assets), "checks": count, "elapsed_seconds": result["elapsed_seconds"]}))


if __name__ == "__main__":
    main()
