"""Reuse existing MACD daily/repair acceptance without new data scans."""
import argparse
import importlib.machinery
import importlib.util
import json
import os

from common import BASE, DAY, HOME, OUT, load, no_conflict, save


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('mode', choices=['daily', 'repair'])
    mode = parser.parse_args().mode
    directory = OUT / 'macd_kdj'
    native = load(directory / f'{mode}-native-run.json')
    path = BASE.parent / 'datasets_recovery_20261028_20261029/macd_readback_20261028.py.txt'
    loader = importlib.machinery.SourceFileLoader('existing_macd_readback', str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    module.B10_DAY = DAY
    module.B10_REPORT = directory
    from bounded_events import CompletionEventReader
    original_current = module.b10_current_factor_status
    def bounded_current(storage, expected):
        return original_current(CompletionEventReader(storage), expected)
    module.b10_current_factor_status = bounded_current
    os.environ['DAGSTER_HOME'] = str(HOME)
    result = module.verify_run(mode, native['run_id'], directory)
    save(directory / f'{mode}-control-after-acceptance.json', no_conflict())
    print(json.dumps({'mode': mode, 'run_id': result['run_id'], 'verified': result['verified'],
                      'checks': len(result['checks']), 'indicator_files': result['indicator_files'],
                      'state_files': result['state_files'], 'elapsed_seconds': result['elapsed_seconds']}))


if __name__ == '__main__':
    main()
