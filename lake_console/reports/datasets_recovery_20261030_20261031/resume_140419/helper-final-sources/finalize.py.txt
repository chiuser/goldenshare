"""One scoped count-only closeout, then restore only the five paused sensors."""
import argparse
import json
from pathlib import Path
from types import SimpleNamespace

from common import BASE, DAY, OUT, ROOT, SENSORS, controls, gql, load, no_conflict, now, require, save, stores

FREQS = (1, 5, 15, 30, 60, 90, 120)


def native_coverage():
    groups={}
    for layer,batch,expected in [('raw','batch1',15),('silver','batch2',20)]:
        old=load(BASE/batch/'daily-run-verified.json')
        new=load(OUT/f'{layer}60/daily-run-verified.json')
        kept=[c for c in old['checks'] if c['asset']!=f'{layer}_stk_mins_60m']
        groups[layer]=kept+new['checks']
        require(len(groups[layer])==expected and len({(c['asset'],c['name']) for c in groups[layer]})==expected,
                'Raw/Silver composite checks incomplete')
    qfq=load(OUT/'qfq-current-seven-accepted.json')
    groups['qfq']=qfq['old_checks']+qfq['new_checks']
    for phase,folder,mode,expected in [('factor','factor','factor',7),
            ('macd_daily','macd_kdj','daily',28),('macd_repair','macd_kdj','repair',14),
            ('turnover','turnover','daily',1)]:
        checks=load(OUT/folder/f'{mode}-run-verified.json')['checks']
        require(len(checks)==expected,'Incorrect original check count '+phase)
        groups[phase]=checks
    groups['nineturn']=load(OUT/'nineturn/final-verified.json')['checks']
    require(len(groups['qfq'])==28 and len(groups['nineturn'])==4,'QFQ/Nine coverage incomplete')
    for phase,checks in groups.items():
        require(all(c['passed'] and c['blocking'] and c['partition'] in (None,DAY) for c in checks),
                'Non-passing/wrong-date original check '+phase)
    total=sum(map(len,groups.values()))
    require(total==117,'Unexpected final logical check coverage')
    return {'total':total,'by_stage':{k:len(v) for k,v in groups.items()}}


def prerequisites():
    for stage, mode in [('raw60','daily'),('silver60','daily'),('qfq60_120','daily'),
                        ('turnover','daily'),('factor','factor'),('macd_kdj','daily'),('macd_kdj','repair')]:
        report = load(OUT / stage / f'{mode}-run-verified.json')
        native = load(OUT / stage / f'{mode}-native-run.json')
        require(report['verified'] and report['stage_acceptance_complete'] and report['date']==DAY
                and report['status']=='SUCCESS' and report['run_id']==native['run_id']
                and native['status']=='SUCCESS', f'Unaccepted stage: {stage}/{mode}')
    require(load(OUT/'qfq-current-seven-accepted.json')['verified'], 'Seven current QFQ missing')
    nine = load(OUT/'nineturn/final-verified.json')
    native = load(OUT/'nineturn/daily-native-run.json')
    require(nine['passed'] and nine['stage_acceptance_complete'] and nine['date']==DAY
            and nine['status']=='SUCCESS' and nine['run_id']==native['run_id']
            and nine['materializations_verified']==4 and nine['blocking_checks_passed']==4,
            'Nine-turn not accepted')
    with stores() as (_, events):
        from orchestrator.defs.asset_guards.stk_mins_qfq_macd_kdj import gold_stk_mins_qfq_macd_kdj_daily_repair_gate_status
        from bounded_events import CompletionEventReader
        reader=CompletionEventReader(events)
        f = SimpleNamespace(event_log_storage=events,get_event_records=reader.get_event_records,
                            fetch_materializations=events.fetch_materializations)
        gate = gold_stk_mins_qfq_macd_kdj_daily_repair_gate_status(f,DAY)
        require(gate.ready and gate.to_payload()==load(OUT/'macd_kdj/repair-run-verified.json')['full_gate'],
                'Final factor/MACD repair gate differs')
    return gate.to_payload()


def count():
    gate = prerequisites()
    coverage = native_coverage()
    pre = no_conflict()
    target = OUT/'final37-physical-count.json'
    require(not target.exists(), 'One final count already exists; inspect rather than repeat')
    from orchestrator.defs.catalog.lake_assets import LAKE_ASSET_CATALOG
    from orchestrator.defs.duckdb_connection import DuckDBConnectionSettings,connect_configured_duckdb
    assets = {f'{layer}_stk_mins_{f}m' for layer in ('raw','silver') for f in (1,5,15,30,60)}
    assets |= {f'gold_stk_mins_qfq_{f}m' for f in FREQS}
    assets |= {f'gold_stk_mins_qfq_macd_kdj_{f}m' for f in FREQS}
    assets |= {f'gold_stk_mins_qfq_macd_kdj_state_{f}m' for f in FREQS}
    assets |= {f'gold_stk_mins_qfq_nineturn_{f}m' for f in (30,60,90,120)}
    assets |= {'gold_wealth_market_turnover','prod_core_wealth_market_turnover'}
    require(len(assets)==37, 'Wrong final count scope')
    catalog = {str(e.asset_key):e for e in LAKE_ASSET_CATALOG}
    result = {'date':DAY,'started_at':now(),'scope':'37 approved minute-chain assets only',
              'count_only':True,'annual_year':2026,'rows':[],'passed':False,
              'factor_macd_full_gate':gate,'controls_before':pre,'global_catalog_audit':False,
              'native_check_coverage':coverage}
    save(target,result)
    try:
        with connect_configured_duckdb(DuckDBConnectionSettings(temp_directory=Path('/private/tmp')),
                                       temp_policy='existing_no_spill') as connection:
            for key in sorted(assets):
                if key=='prod_core_wealth_market_turnover':
                    native = load(OUT/'turnover/daily-native-run.json')
                    logs = list((Path('/Users/congming/.goldenshare/dagster_logs')/native['run_id']/'compute_logs').glob('*.out'))
                    matching=[line for p in logs for line in p.read_text().splitlines()
                              if 'event=prod_core_wealth_market_turnover_completed' in line and f'partition_key={DAY}' in line]
                    require(len(matching)==1 and 'output_row_count=5' in matching[0] and 'read_back_row_count=5' in matching[0],
                            'Missing original Prod count readback')
                    result['rows'].append({'asset':key,'row_count':5,'status':'present',
                       'evidence':'original Prod job exact-date write/readback count; no redundant DB query',
                       'run_id':native['run_id']})
                    continue
                entry=catalog[key]
                require(entry.path_template.startswith('data_lake/'),'Unexpected formal path')
                pattern=str(ROOT/entry.path_template.removeprefix('data_lake/').format(
                    partition_key=DAY,ts_code='*',year='2026'))
                annual='ts_code=*' in pattern
                if not annual:require(Path(pattern).is_file(),'Missing direct partition '+key)
                query='SELECT count(*) FROM read_parquet(?,hive_partitioning=false)'
                parameters=[pattern]
                if annual:
                    query+=' WHERE trade_date=CAST(? AS DATE)';parameters.append(DAY)
                rows=int(connection.execute(query,parameters).fetchone()[0])
                require(rows>0,'Zero target-date count '+key)
                result['rows'].append({'asset':key,'row_count':rows,'status':'present',
                                       'path_or_pattern':pattern,'date_filter':DAY if annual else None})
        result.update(passed=True,complete_asset_count=37,missing_asset_count=0,
                      current_native_check_coverage=coverage['total'],completed_at=now())
    finally:save(target,result)
    print(json.dumps({'date':DAY,'assets':37,'missing':0,'passed':result['passed'],
                      'native_check_coverage':117}))


def restore():
    audit=load(OUT/'final37-physical-count.json')
    require(audit['passed'] and audit['complete_asset_count']==37 and audit['missing_asset_count']==0,
            'Final count not complete')
    prerequisites()
    before=no_conflict()
    original={x['name']:x for x in load(BASE/'sensor-hold-after-failure.json')['sensors']}
    result={'date':DAY,'started_at':now(),'before':before,'responses':[],
            'cursor_mutations':0,'passed':False,'restored_count':0}
    destination=OUT/'sensor-restoration.json'
    require(not destination.exists(),'Existing restoration must be inspected')
    save(destination,result)
    for name in SENSORS:
        selector={'repositoryLocationName':'orchestrator','repositoryName':'__repository__','sensorName':name}
        value=gql('mutation S($s:SensorSelector!){startSensor(sensorSelector:$s){__typename ... on Sensor{name sensorState{id status typeSpecificData{... on SensorData{lastCursor lastRunKey}}}}}}',{'s':selector})['startSensor']
        result['responses'].append({'at':now(),'selector':selector,'response':value})
        save(destination,result)
        require(value['__typename']=='Sensor' and value['name']==name and value['sensorState']['status']=='RUNNING',
                'Sensor start not confirmed')
        state=value['sensorState'];detail=state['typeSpecificData'];saved=original[name]
        require(state['id'].split('::')[-1]==saved['selector_id'],'Sensor identity changed')
        require(detail['lastCursor']==saved['cursor'] and detail['lastRunKey']==saved['last_run_key'],
                'Immediate cursor/key changed; inspect before any further mutation')
        result['restored_count']+=1
    after=controls()
    require(all(x['sensorState']['status']=='RUNNING' for x in after['sensors']),'Five sensors not running')
    require(not after['conflicting_runs'],'New minute-chain activity requires inspection')
    result.update(after=after,passed=True,completed_at=now())
    save(destination,result)
    print(json.dumps({'date':DAY,'restored':5,'cursor_mutations':0,'passed':True}))


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('mode',choices=['count','restore'])
    {'count':count,'restore':restore}[p.parse_args().mode]()
