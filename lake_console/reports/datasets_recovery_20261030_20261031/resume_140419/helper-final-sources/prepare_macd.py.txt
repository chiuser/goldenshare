"""Necessary original guards and metadata budgets; no source-price calculation."""
import argparse
import json
from dataclasses import asdict
from pathlib import Path
from types import SimpleNamespace

from common import BASE, DAY, OUT, ROOT, load, no_conflict, now, require, save, stores

PRIOR = "2026-10-29"
FREQS = (1, 5, 15, 30, 60, 90, 120)
DIRECTORY = OUT / "macd_kdj"
GIB = 1024 ** 3


def facade(events):
    return SimpleNamespace(event_log_storage=events, get_event_records=events.get_event_records,
                           fetch_materializations=events.fetch_materializations)


def stat_group(paths):
    paths = list(paths)
    return {"files": len(paths), "bytes": sum(p.stat().st_size for p in paths)}


def daily():
    from orchestrator.defs.asset_guards.stk_mins_qfq_factor_repair import gold_stk_mins_qfq_factor_repair_status
    from orchestrator.defs.paths import gold_stk_mins_qfq_macd_kdj_state_path
    from orchestrator.defs.sensors.gold_stk_mins_qfq_macd_kdj_daily_update_job_sensor import GOLD_STK_MINS_QFQ_MACD_KDJ_STATE_READINESS_SPECS
    from orchestrator.defs.sensors.readiness import partition_dataset_readiness_status_from_latest_checks
    factor = load(OUT / "factor/factor-run-verified.json")
    require(factor["verified"] and factor["stage_acceptance_complete"], "Factor not accepted")
    control = no_conflict()
    with stores() as (_, events):
        f = facade(events)
        gate = gold_stk_mins_qfq_factor_repair_status(f, DAY)
        require(gate.ready and gate.to_payload() == factor["factor_gate"], "Accepted factor identity changed")
        previous = partition_dataset_readiness_status_from_latest_checks(
            f, GOLD_STK_MINS_QFQ_MACD_KDJ_STATE_READINESS_SPECS, partition_key=PRIOR)
        require(previous.ready, "Previous29 state native checks not ready")
    states = {}
    for freq in FREQS:
        path = gold_stk_mins_qfq_macd_kdj_state_path(ROOT, freq, PRIOR)
        value = path.stat()
        states[str(path)] = [value.st_dev, value.st_ino, value.st_size, value.st_mtime_ns, value.st_ctime_ns]
        require(not gold_stk_mins_qfq_macd_kdj_state_path(ROOT, freq, DAY).exists(), "Current-day state exists; inspect completed work")
    # Metadata only. These exact annual groups are the approved daily inputs.
    groups = {}
    for year in (2025, 2026):
        groups[f"qfq_{year}"] = stat_group(p for freq in FREQS for p in
            (ROOT / 'gold/quote/stk_mins_qfq' / f'freq={freq}').glob(f'ts_code=*/year={year}/part-000.parquet'))
    groups['indicator_2026'] = stat_group(p for freq in FREQS for p in
        (ROOT / 'gold/indicator/stk_mins_qfq_macd_kdj' / f'freq={freq}').glob('ts_code=*/year=2026/part-000.parquet'))
    qy, qp, old_i = (groups[key]['bytes'] / GIB for key in ('qfq_2026','qfq_2025','indicator_2026'))
    v = sum(value[2] for value in states.values()) / GIB
    # Original LLD19.7 formula; L<=Qp, C<=22GiB, P<=0.125GiB, Fr/Fw=1, Tr/Tw=8.
    read = 6*qy + 5*qp + 2*old_i + 4*22 + 0.125 + v + 1 + 8
    write = 22 + 0.125 + 1 + 8
    budget = {'Qy': qy, 'Qp': qp, 'L_upper': qp, 'I': old_i, 'C_estimate': 22,
              'P_estimate': 0.125, 'V': v, 'footer_reserve': 1, 'engine_work_reserve': 8,
              'read_estimate_gib': read, 'write_estimate_gib': write,
              'approved_daily_read_gib': 248, 'approved_daily_write_gib': 36}
    save(DIRECTORY / 'current-metadata-budget.json', {'observed_at': now(), 'groups': groups, 'budget': budget, 'price_content_reads': 0})
    require(read <= 248 and write <= 36, 'Current metadata no longer fits the approved daily budget; stop before launch')
    pre = {'date': DAY, 'safe_to_launch': True, 'observed_at': now(), 'factor_gate': gate.to_payload(),
           'upstream_factor_run_id': factor['run_id'], 'previous_state_inputs': states,
           'previous_state_native_readiness': asdict(previous), 'budget': budget,
           'controls': control, 'formal_writes': 0, 'price_content_reads': 0}
    save(DIRECTORY / 'preflight.json', pre)
    save(DIRECTORY / 'run-config.json', {})
    print(json.dumps({'macd_daily_ready': True, 'actual_repair_count': gate.repair_required_code_count,
                      'estimated_read_gib': read, 'estimated_write_gib': write}))


def repair():
    from orchestrator.defs.asset_guards.stk_mins_qfq_factor_repair import gold_stk_mins_qfq_factor_repair_status
    from orchestrator.defs.asset_guards.stk_mins_qfq_macd_kdj import gold_stk_mins_qfq_macd_kdj_daily_repair_gate_status, gold_stk_mins_qfq_macd_kdj_repair_completion_status_for_upstream_batch
    from orchestrator.defs.sensors.gold_stk_mins_qfq_macd_kdj_daily_update_job_sensor import GOLD_STK_MINS_QFQ_MACD_KDJ_READINESS_SPECS
    from orchestrator.defs.sensors.gold_stk_mins_qfq_macd_kdj_repair_job_sensor import build_gold_stk_mins_qfq_macd_kdj_repair_run_status_decision, _run_config_for_repair_decision
    from orchestrator.defs.sensors.readiness import partition_dataset_readiness_status_from_latest_checks
    daily_result = load(DIRECTORY / 'daily-run-verified.json')
    require(daily_result['verified'] and daily_result['stage_acceptance_complete'], 'Daily not accepted')
    control = no_conflict()
    with stores() as (_, events):
        f = facade(events)
        factor = gold_stk_mins_qfq_factor_repair_status(f, DAY)
        require(factor.ready and factor.to_payload() == daily_result['factor_gate'], 'Factor identity changed')
        readiness = partition_dataset_readiness_status_from_latest_checks(f,
            GOLD_STK_MINS_QFQ_MACD_KDJ_READINESS_SPECS, partition_key=DAY)
        require(readiness.ready and {s.asset_key:s.materialization_storage_id for s in readiness.statuses}
                == {m['asset']:m['storage_id'] for m in daily_result['materializations']}, 'Daily current checks differ from accepted materializations')
        decision = build_gold_stk_mins_qfq_macd_kdj_repair_run_status_decision(
            target_trade_date=DAY, qfq_factor_repair_status=factor, macd_kdj_daily_ready=readiness.ready)
        result = {'date': DAY, 'observed_at': now(), 'factor_gate': factor.to_payload(),
                  'daily_run_id': daily_result['run_id'], 'decision': asdict(decision),
                  'repair_to_launch': False, 'run_config': None, 'controls': control,
                  'formal_writes':0, 'price_content_reads':0}
        if not factor.rewrote_history:
            gate = gold_stk_mins_qfq_macd_kdj_daily_repair_gate_status(f, DAY)
            require(gate.ready and not gate.requires_macd_kdj_repair, 'No-repair gate not ready')
            result['full_gate'] = gate.to_payload()
        else:
            completion = gold_stk_mins_qfq_macd_kdj_repair_completion_status_for_upstream_batch(f,
                qfq_factor_repair_trade_date=DAY, repair_start_trade_date=factor.repair_start_trade_date,
                repair_end_trade_date=factor.repair_end_trade_date, upstream_batch_id=factor.upstream_batch_id,
                repair_required_code_count=factor.repair_required_code_count,
                repair_required_codes_hash=factor.repair_required_codes_hash)
            if completion.ready:
                gate = gold_stk_mins_qfq_macd_kdj_daily_repair_gate_status(f, DAY)
                require(gate.ready, 'Existing repair has no complete gate')
                result['full_gate'] = gate.to_payload()
            else:
                require(decision.selected_trade_date is not None, decision.reason)
                config = _run_config_for_repair_decision(decision)
                result.update(repair_to_launch=True, run_config=config)
                require(config['ops']['gold_stk_mins_qfq_macd_kdj_repair_op']['config']['stock_codes']
                        == factor.to_payload()['repair_required_codes'], 'Original repair config scope mismatch')
                save(DIRECTORY / 'repair-run-config.json', config)
    save(DIRECTORY / 'repair-decision.json', result)
    print(json.dumps({'repair_to_launch': result['repair_to_launch'], 'actual_repair_count': result['factor_gate']['repair_required_code_count']}))


if __name__ == '__main__':
    p=argparse.ArgumentParser();p.add_argument('mode',choices=['daily','repair'])
    {'daily':daily,'repair':repair}[p.parse_args().mode]()
