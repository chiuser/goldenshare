"""Reuse original factor receipt/gate verifier; do not generate repair lists."""
import json
from pathlib import Path

import dagster as dg

from common import DAY, OUT, load, no_conflict, now, require, save, stores
from accept_simple import helper


def main():
    directory = OUT / "factor"
    native = load(directory / "factor-native-run.json")
    rid = native["run_id"]
    module = helper()
    with stores() as (rs, es):
        found = rs.get_run_records(dg.RunsFilter(run_ids=[rid]), limit=1)
        require(len(found) == 1, "Missing exact factor run")
        record = found[0]
        run = record.dagster_run
        require(run.status == dg.DagsterRunStatus.SUCCESS, "Factor run unsuccessful")
        mats, checks, failures = module.events_for_run(es, rid)
        require(not failures, "Factor failure event")
        md, gate = module.verify_factor(run, mats, checks, es)
        require(gate["repair_start_trade_date"] == "2014-01-02"
                and gate["repair_end_trade_date"] == DAY
                and gate["selected_partition_count"] == 3117, "Original repair scope differs from approved range")
        logs = list((Path('/Users/congming/.goldenshare/dagster_logs') / rid / "compute_logs").glob("*.out"))
        require(logs, "Missing factor logs")
        progress, targets, sources, factors, as_of = module.load_logs(logs)
        publications = [module.verify_operation(d, value, run.root_run_id or rid, "factor", md)
                        for d, value in progress.items()]
        source_accounting = module.verify_factor_logs(md, publications, targets, sources, factors, as_of)
        result = {"date": DAY, "mode": "factor", "run_id": rid,
                  "root_run_id": run.root_run_id or rid, "job": run.job_name,
                  "status": "SUCCESS", "verified": True, "stage_acceptance_complete": True,
                  "accepted_at": now(), "elapsed_seconds": record.end_time - record.start_time,
                  "materializations": mats, "checks": checks, "factor_gate": gate,
                  "publications": publications, "factor_source_accounting": source_accounting,
                  "formal_writes_by_readback": 0, "extra_price_reads": 0}
    save(directory / "factor-run-verified.json", result)
    save(directory / "control-after-acceptance.json", no_conflict())
    print(json.dumps({"factor_run": rid, "verified": True, "checks": len(checks),
                      "actual_repair_count": gate["repair_required_code_count"],
                      "start": gate["repair_start_trade_date"], "end": gate["repair_end_trade_date"],
                      "elapsed_seconds": result["elapsed_seconds"]}))


if __name__ == "__main__":
    main()
