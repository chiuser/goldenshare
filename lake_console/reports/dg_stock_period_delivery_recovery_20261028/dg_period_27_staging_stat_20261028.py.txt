from pathlib import Path
from datetime import datetime
import json
R=Path('/Users/congming/github/goldenshare/lake_console/reports/dg_stock_period_delivery_recovery_20261028')

summary={};parquet=[]
for p in W.rglob('*'):
 if not p.is_file():continue
 relative=p.relative_to(W);category=relative.parts[0];x=summary.setdefault(category,{'files':0,'bytes':0});x['files']+=1;x['bytes']+=p.stat().st_size
 if p.suffix=='.parquet':parquet.append({'path':str(relative),'bytes':p.stat().st_size})
units=[]

(R/'raw27-week-staging-stat.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k not in ('parquet_file_attribute_facts','successful_source_units')},ensure_ascii=False,indent=2))
