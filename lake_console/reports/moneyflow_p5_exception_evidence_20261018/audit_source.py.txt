def read(path):
    global bytes_read
    p = ROOT / path
    assert not Path(path).is_absolute() and '..' not in Path(path).parts
    assert p.resolve().is_relative_to(ROOT) and p.is_file()
    if path not in cache:
        size = p.stat().st_size
        assert size <= 16 * 1024**2 and len(cache) < 1000
        bytes_read += size
        assert bytes_read <= 128 * 1024**2
        cache[path] = p.read_bytes()
    return cache[path]

def sha(path):
    return hashlib.sha256(read(path)).hexdigest()

def doc(path):
    return json.loads(read(path))

pinned = {}
def pin(path, expected):
    assert sha(path) == expected, f'evidence/source changed: {path}'
    if path in pinned:
        assert pinned[path] == expected
    pinned[path] = expected

def tests(module, function, exact=None):
    path = PROJ + 'tests/' + module + '.py'
    parsed = ast.parse(read(path))
    functions = [n for n in ast.walk(parsed) if isinstance(n, ast.FunctionDef) and n.name == function]
    assert len(functions) == 1, (module, function)
    matches = [{'report': report, 'module': m, 'testcase': name,
                'test_file': path, 'line': functions[0].lineno}
               for report, m, name in cases if m == module and
               (name == exact if exact else name.split('[')[0] == function)]
    assert matches, (module, function, exact)
    return matches
