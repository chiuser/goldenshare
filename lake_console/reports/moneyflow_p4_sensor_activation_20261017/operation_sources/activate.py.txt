def check_ticks(states,ds):
 completed={}
 for n,x in states.items():
  assert x['sensorState']['status']=='RUNNING','sensor_not_running'
  epoch=read_json(action_dir(n)/'start-intent.json')['epoch']
  candidates=[t for t in x['sensorState']['ticks'] if t['timestamp']>=epoch]
  for t in candidates:
   assert t['status']!='FAILURE' and not t['error'],'sensor_tick_failed'
   assert not t['runIds'] and not t['runKeys'] and not t['dynamicPartitionsRequestResults'],'unexpected_business_action'
  done=next((t for t in candidates if t['status'] in ['SUCCESS','SKIPPED'] and t['endTimestamp']),None)
  if done is None:continue
  assert done['cursor'] and len(done['cursor'].encode())<=8192,'cursor_missing_or_large'
  cursor=load_sensor_cursor(done['cursor']);details=cursor['details'];assert details['sensor_name']==n and details['asset_family']==ds and details['partition_set']==CONTRACTS[ds].partition_set,'cursor_identity'
  if not n.startswith('cn_a_'):assert details['reason_code']=='all_ready','update_not_ready'
  else:assert cursor['selected_count']==0,'unexpected_registration'
  completed[n]={'tick':done,'cursor':cursor,'actual_status':x['sensorState']['status']}
 return completed
