def names(ds):
 c=CONTRACTS[ds]
 return ['cn_a_'+ds+'_trade_day_sensor',c.silver_job+'_sensor',c.raw_job+'_sensor']
NAMES=[n for ds in DATASETS for n in names(ds)]
def selector(name):
 assert name in NAMES,'sensor_not_authorized'
 return {**REPOSITORY,'sensorName':name}
def schema_check(body):
 schema=create_schema().graphql_schema;doc=parse(body['query']);assert not validate(schema,doc),'graphql_schema';assert isinstance(get_variable_values(schema,doc.definitions[0].variable_definitions,body.get('variables',{})),dict),'graphql_variables'
def fetch(b,chosen):
 assert chosen and set(chosen)<=set(NAMES)
 variables={f's{i}':selector(n) for i,n in enumerate(chosen)};decl=','.join('$'+k+':SensorSelector!' for k in variables)
 query='query('+decl+'){'+''.join(f'a{i}:sensorOrError(sensorSelector:$s{i}){{{FIELDS}}}' for i in range(len(chosen)))+'}'
 data=b.http({'query':query,'variables':variables})['data'];result={}
 for i,n in enumerate(chosen):
  x=data[f'a{i}'];assert x['__typename']=='Sensor' and x['name']==n and x['defaultStatus']=='STOPPED' and x['minIntervalSeconds']==600
  expected=[] if n.startswith('cn_a_') else [{'pipelineName':n[:-7]}]
  assert x['targets']==expected,(n,'loaded_target')
  result[n]=x
 return result
