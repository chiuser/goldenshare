            yield session
            control.check_active()
        except (SQLAlchemyError, PsycopgError) as error:
            control.check_active()  # 已取消/到期时抛原控制分类
            raise SourceReadError(
                "stock_daily_database/" + phase + "/" + type(error).__name__
            ) from error
    except BaseException as error:
        primary = error
        raise
    finally:
        cleanup_notes = []
        first_cleanup_error = None
        invalidate_attempted = False

        def record_cleanup(stage, error):
            nonlocal first_cleanup_error
            cleanup_notes.append(stage + ":" + type(error).__name__)
            if primary is None and first_cleanup_error is None:
                first_cleanup_error = error

        def discard_connection():
            nonlocal invalidate_attempted
            if invalidate_attempted:
                return
            invalidate_attempted = True
            try:
                if connection is not None:
                    if not connection.closed:
                        connection.invalidate()
                elif session is not None:
                    session.invalidate()
            except BaseException as error:
                record_cleanup("invalidate", error)

        try:
            if connection is not None and listener_attempted:
                try:
                    if event.contains(connection, "before_cursor_execute", set_timeout):
                        event.remove(connection, "before_cursor_execute", set_timeout)
                except BaseException as error:
                    record_cleanup("remove_listener", error)
                    discard_connection()
            if session is not None:
                try:
                    session.close()  # 正常资源归还唯一入口，不commit
                except BaseException as error:
                    record_cleanup("close_session", error)
                    discard_connection()
                    if connection is not None:
                        try:
                            connection.close()
                        except BaseException as error:
                            record_cleanup("close_connection", error)
        finally:
            session = connection = None

        if cleanup_notes:
            if primary is not None:
                primary.add_note("stock_daily_database cleanup: " + ", ".join(cleanup_notes))
            elif not isinstance(first_cleanup_error, Exception):
                raise first_cleanup_error
            else:
                raise ResourceCleanupError(
                    "stock_daily_database cleanup: " + ", ".join(cleanup_notes)
                ) from first_cleanup_error

监听器尝试登记前即标记；登记中途报错，清理仍检查本Connection是否已有同一函数，不按“未返回就一定没登记”推测。event.contains/remove仅在业务SQL已经退出的该工作线程调用，不能在监听器自己执行时移除，也不访问全局Engine事件。移除失败时先废弃连接再关闭Session；Session.close失败时废弃尚未安全归还的连接，并尝试Connection.close。每项实际清理仅尝试一次，前一项失败不会跳过后一项，不重试业务查询。

-- 独立21行参考：ref_prices只含seq及显式字面close_cents/high_cents/low_cents。
-- 前10行500/550/450，后11行1000/1100/900。先建表插入21行，再仅执行本查询一次。
WITH RECURSIVE
b AS MATERIALIZED (
  SELECT seq, close_cents::HUGEINT * 10000000000000000::HUGEINT AS x,
         close_cents,
         max(high_cents) OVER (ORDER BY seq ROWS BETWEEN 8 PRECEDING AND CURRENT ROW) AS hi,
         min(low_cents) OVER (ORDER BY seq ROWS BETWEEN 8 PRECEDING AND CURRENT ROW) AS lo
  FROM ref_prices
),
v AS MATERIALIZED (
  SELECT *, CASE WHEN hi=lo THEN 50000000000000000000::HUGEINT
    ELSE ((close_cents-lo)::HUGEINT * 100::HUGEINT *
          1000000000000000000::HUGEINT) // (hi-lo) END AS rsv FROM b
),
m(seq,ef,es,dif,dea) AS (
  SELECT seq,x,x,0::HUGEINT,0::HUGEINT FROM b WHERE seq=1
  UNION ALL
  SELECT b.seq,e.ef,e.es,e.ef-e.es,
         (8::HUGEINT*m.dea+2::HUGEINT*(e.ef-e.es))//10
  FROM m JOIN b ON b.seq=m.seq+1
  CROSS JOIN LATERAL (
    SELECT (11::HUGEINT*m.ef+2::HUGEINT*b.x)//13 AS ef,
           (25::HUGEINT*m.es+2::HUGEINT*b.x)//27 AS es
  ) e
),
k(seq,kval,dval) AS (
  SELECT seq,first_k,(100000000000000000000::HUGEINT+first_k)//3
  FROM v CROSS JOIN LATERAL (
    SELECT (100000000000000000000::HUGEINT+v.rsv)//3 AS first_k
  ) init WHERE seq=1
  UNION ALL
  SELECT v.seq,next_k,(2::HUGEINT*k.dval+next_k)//3
  FROM k JOIN v ON v.seq=k.seq+1
  CROSS JOIN LATERAL (SELECT (2::HUGEINT*k.kval+v.rsv)//3 AS next_k) step
),
w AS (
  SELECT seq,
    CASE WHEN count(*) OVER w5=5 THEN sum(close_cents) OVER w5/500.0 END AS ma5,
    CASE WHEN count(*) OVER w10=10 THEN sum(close_cents) OVER w10/1000.0 END AS ma10,
    CASE WHEN count(*) OVER w20=20 THEN sum(close_cents) OVER w20/2000.0 END AS middle,
    CASE WHEN count(*) OVER w20=20 THEN sqrt(
      (20::HUGEINT*sum(close_cents::HUGEINT*close_cents) OVER w20 -
       (sum(close_cents) OVER w20)::HUGEINT*(sum(close_cents) OVER w20)::HUGEINT)
       ::DOUBLE/4000000.0) END AS deviation
  FROM ref_prices
  WINDOW w5 AS (ORDER BY seq ROWS BETWEEN 4 PRECEDING AND CURRENT ROW),
         w10 AS (ORDER BY seq ROWS BETWEEN 9 PRECEDING AND CURRENT ROW),
         w20 AS (ORDER BY seq ROWS BETWEEN 19 PRECEDING AND CURRENT ROW)
)
SELECT m.seq, m.dif::DOUBLE/1e18 AS dif, m.dea::DOUBLE/1e18 AS dea,
       (2::HUGEINT*(m.dif-m.dea))::DOUBLE/1e18 AS hist,
       k.kval::DOUBLE/1e18 AS k, k.dval::DOUBLE/1e18 AS d,
       (3::HUGEINT*k.kval-2::HUGEINT*k.dval)::DOUBLE/1e18 AS j,
       w.ma5,w.ma10,w.middle AS ma20,w.middle,
       w.middle+2*w.deviation AS upper,w.middle-2*w.deviation AS lower
FROM m JOIN k USING(seq) JOIN w USING(seq) ORDER BY m.seq;

参考SQL使用HUGEINT定点递推，最大中间数量远低于HUGEINT范围；每次整数除法的舍弃量在本21行样本中远小于既有1e-8误差门槛。参考读取21行不等于免费SQL：窗口、递推和合并各阶段逻辑量按固定步骤逐项登记，不以任意倍率或固定余量代替。先核字面锚点：MACD首根三值0、K/D/J首根50；MA20第20根7.5、第21根7.75；BOLL第20根12.5/7.5/2.5。若参考与这些可手算值冲突，停止修正参考依据，不运行产品节点去迎合预期。本轮文档审计未执行此SQL，其首次实际执行属于PREP，失败同样停下。

  1. Reader首次使用时列举本次截止日前的分区名称，保持逆序的只读日期清单；不跨请求缓存，不把清单传到内核或让业务指定文件路径。文件名日期只用于定位，返回trade_date始终取文件列。当页有start_date时，候选日期早于start_date立即结束当页定位，不能为凑limit继续扫描整个旧历史；该较早位置留给后续预热使用。
  2. 每轮取不超过“本次还缺的根数”的候选文件，且最多config.batch_rows份；使用PyArrow23.0.1显式文件列表Dataset/Scanner，按ts_code及请求日期过滤并只投影所需列；以生产者合同定义本地13列/4列schema，不再单独打开文件反复检查schema或读取全宽字段。这些列只是本地读取投影，不新建Tushare字段子契约，也不修改既有完整接口合同。Scanner的batch_size沿config.batch_rows，batch_readahead=0、fragment_readahead=0、use_threads=False、cache_metadata=False，Parquet扫描pre_buffer=False；逐批前后check_active，在过滤后的结果上转行，不把全市场Table转成Python对象。
  3. 一个证券每天至多一行由外部合同保证。某批文件筛不到该证券或因停牌根数不满，继续尚未读过的更早文件，直到得到limit根或候选耗尽；不能因为一个文件/一个Arrow批为空就宣称已经到头。已取记录只在本次内存中按trade_date升序整理一次。
  4. 当页读取用13列；预热仅用4列。Reader保留自己的候选进度，预热从当页最早日期之前继续，既不重新枚举全清单，也不重复打开已经消费过的当页文件。请求start_date只限制返回页，不能限制更早预热；预热不按自然日倒推，不用交易日历补停牌K线。

19.5 一次请求的固定执行顺序与释放

def work(self, request, control):
    reader = snapshot = price_source = result = dto = payload = None
    session = load_rows = read_previous = page_prices = calculation = None
    primary_error = None
    cleanup_errors = ()
    try:
        control.check_active()
        if self._use_dg:
            reader = StockDailyLakeReader(self.runtime.config)
        with open_stock_daily_read_session(self.session_factory_provider(), control) as session:
            def load_rows(code, end, start, count):
                if reader is not None:
                    return reader.load_kline_rows(code, end, start, count, control.check_active)
                return self._price_query.load_kline_rows(
                    session, ts_code=code, end_date=end, start_date=start, limit=count,
                )
            snapshot = self.service.prepare(session, request, load_rows=load_rows)
        session = load_rows = None
        control.check_active()
        if snapshot.bars:
            def read_previous(before, count):
                if reader is not None:
                    values = reader.load_previous_kline_rows(
                        snapshot.ts_code, before, count, control.check_active,
                    )
                else:
                    with open_stock_daily_read_session(self.session_factory_provider(), control) as prior:
                        values = self._price_query.load_previous_kline_rows(
                            prior, ts_code=snapshot.ts_code, before=before, limit=count,
                        )
                return tuple({"trade_date": r["trade_date"], "high": r["high_qfq"],
                              "low": r["low_qfq"], "close": r["close_qfq"]} for r in values)
            page_prices = tuple({"trade_date": r.trade_date, "high": r.high_qfq,
                                 "low": r.low_qfq, "close": r.close_qfq} for r in snapshot.bars)
            price_source = StockDailyPriceSource(
                page_prices, read_previous, source_id=self._price_source_id,
            )
            page_prices = None
            calculation = build_calculation_request(snapshot)
            result = TechnicalIndicatorService(
                source=price_source, registry=self.runtime.registry,
                config=self.runtime.config, control=control,
            ).calculate(calculation)
            calculation = read_previous = price_source = None
        dto = self.service.assemble(snapshot, result, source_tables=self._source_tables)
        snapshot = result = None
        payload = dto.model_dump(mode="json", by_alias=True,
                                 exclude_none=False, exclude_unset=False)
        dto = None
        control.check_active()
        return payload
    except BaseException as error:
        primary_error = error
        raise
    finally:
        if reader is not None:
            try:
                reader.close()
